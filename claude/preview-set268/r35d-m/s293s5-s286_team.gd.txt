func _fit2_w(hv: Array) -> float:   # s287: narrowest width that holds every value in 2 rows
	var ws := []
	for it in hv: ws.append(_chip_w(it[0], it[1]))
	var best := 0.0
	if ws.size() > 0:
		best = 1e9
		for k in range(ws.size() + 1):
			var a := 0.0
			var b := 0.0
			for i in ws.size():
				if i < k: a += ws[i] + (VG if i > 0 else 0.0)
				else: b += ws[i] + (VG if i > k else 0.0)
			best = minf(best, maxf(a, b))
	return 27.0 + best + 4.0 + 1.0

func _process(_d: float) -> void:
	if view == null or not is_visible_in_tree():
		return
	if _order.is_empty() or _order.size() != _own_count() or _forder.size() != view.units.size() - _own_count():
		_sort_units()
	if not _pre and not _order.is_empty() and OS.get_environment("S286PICK") != "": # stills: pick card N (or f<N> = foe N)
		_pre = true
		var pk := OS.get_environment("S286PICK")
		if pk.begins_with("f"):
			var foes := []
			for v in view.units.values():
				if int(v["side"]) == 1: foes.append(int(v["uid"]))
			foes.sort()
			if not foes.is_empty(): picked = int(foes[clampi(int(pk.substr(1)), 0, foes.size() - 1)])
		else:
			picked = int(_order[clampi(int(pk), 0, _order.size() - 1)])
	_hov287 = -1
	if OS.get_environment("S287HOVER") != "" and _order.size() > int(OS.get_environment("S287HOVER")): _hov287 = int(_order[int(OS.get_environment("S287HOVER"))])
	elif get_viewport() != null:
		var hu := _unit_at(get_viewport().get_mouse_position())
		if hu >= 0 and view.units.has(hu) and int(view.units[hu]["side"]) == 0: _hov287 = hu
	_sync_outline()
	queue_redraw()

func _own_count() -> int:
	var n := 0
	for v in view.units.values():
		if int(v["side"]) == 0: n += 1
	return n

func _grp(uid: int) -> Variant:
	if stage == null: return null
	var gb: Dictionary = stage.groups.get("battle", {})
	var key: Variant = uid if gb.has(uid) else float(uid)
	return gb.get(key)

func _spr(uid: int) -> Sprite3D:
	var g: Variant = _grp(uid)
	if g == null: return null
	var s: Variant = g.get("spr")
	return s as Sprite3D if s is Sprite3D else null

func _sort_units() -> void:
	_cw.clear()
	_seen.clear()
	_mx0.clear()
	_order = _side_order(0)
	_forder = _side_order(1)   # s287e: foe cards, same order rule

func _side_order(side: int) -> Array:
	var cam := get_viewport().get_camera_3d()
	var rows := []
	for v in view.units.values():
		if int(v["side"]) != side: continue
		var x := float(int(v["lane"]))
		var yy := float(int(v["row"]))
		var s := _spr(int(v["uid"]))
		if cam != null and s != null:
			var sp2 := cam.unproject_position(s.global_position)
			x = sp2.x; yy = sp2.y
		rows.append([x, int(v["uid"]), yy])
	# s287b: card order follows the board slots and never changes: top row left -> right, then the bottom row
	var ys := rows.map(func(r): return r[2]); ys.sort()
	var mid := (float(ys[0]) + float(ys[ys.size() - 1])) * 0.5 if not ys.is_empty() else 0.0
	rows.sort_custom(func(a, b):
		var ra := 0 if float(a[2]) < mid - 1.0 else 1
		var rb := 0 if float(b[2]) < mid - 1.0 else 1
		return ra < rb if ra != rb else a[0] < b[0])
	return rows.map(func(r): return r[1])

func _draw() -> void:
	_rects.clear()
	_ic_q.clear()
	_tips.clear()
	if _ic_layer != null: _ic_layer.queue_redraw()
	if view == null: return
	_bar287.clear()
	_panel_q.clear()
	for v in view.units.values():
		_feet_bars(v)
	if _order.is_empty(): return
	var n := _order.size()
	# r35j: one size for every card: fixed width, height = rows the busiest card needs (min 2), cards packed from the left
	# s287: one width for the whole team = the narrowest card where the busiest card still fits in 2 rows (no "+N");
	# grows when a card gains values, never shrinks during the fight
	# s287b: values in a 2-column grid, 3 rows, filled left then right, row by row; column 2 starts after the widest
	# left value of the team (grows only), so every card lines up
	for uid0 in _order + _forder:
		if not view.units.has(uid0): continue
		var sl0 := _slots(view.units[uid0])
		for k0 in sl0.size():
			var cw0 := _chip_w(sl0[k0][0], sl0[k0][1])
			if k0 % 2 == 0: _c1w287 = maxf(_c1w287, cw0)
			else: _c2w287 = maxf(_c2w287, cw0)
	_tw287 = 27.0 + _c1w287 + (VG + _c2w287 if _c2w287 > 0.0 else 0.0) + 5.0
	var w := minf(maxf(_tw287, CARD_MIN_W), (size.x - PAD_L - PAD_R - GAP * float(n - 1)) / float(n))
	if OS.get_environment("S287FIXW") == "1": w = minf(CARD_W, (size.x - PAD_L - PAD_R - GAP * float(n - 1)) / float(n))
	var rows := 3   # s287b: fixed 3 rows x 2 values (armour moved above the card, max 6 values inside)
	var h := 5.0 + float(rows) * ROW_H
	var y := size.y - PAD_B - h
	for i in n:
		var uid: int = _order[i]
		if not view.units.has(uid): continue
		var r := Rect2(Vector2(PAD_L + float(i) * (w + GAP), y), Vector2(w, h))
		if OS.get_environment("S287CARDS") != "0":
			_rects[uid] = r
			_card(r, view.units[uid], uid == picked)
		elif _bar287.has(uid):
			_attached(view.units[uid], _bar287[uid])
	if OS.get_environment("S287CARDS") != "0":   # s287e: foe cards top-right, same card, chips BELOW the card
		var nf := _forder.size()
		var xf := size.x - 16.0 - float(nf) * w - GAP * float(maxi(nf - 1, 0))
		for i in nf:
			var fu: int = _forder[i]
			if not view.units.has(fu): continue
			var fr2 := Rect2(Vector2(xf + float(i) * (w + GAP), 8.0), Vector2(w, h))
			_rects[fu] = fr2
			_card(fr2, view.units[fu], fu == picked, true)
	if _pn_layer != null: _pn_layer.queue_redraw()

func _dtxt(p: Vector2, t: String, fs: int, col: Color, bold := false, right := -1.0) -> void:

	var x := p.x

func _panel(v: Dictionary, br: Rect2) -> void:   # s287d (owner C1): small stats panel beside the HP bar, tap / hover only

	for k in hv.size():
		if k % 2 == 0: c1 = maxf(c1, _chip_w(hv[k][0], hv[k][1]))
		else: c2 = maxf(c2, _chip_w(hv[k][0], hv[k][1]))

	var rows := (hv.size() + 1) / 2
	var h := 4.0 + 11.5 + float(rows) * 11.5 + 2.0
	var p := Vector2(br.end.x + 5.0, br.end.y - h)
	if p.x + w > size.x - 4.0: p.x = br.position.x - 5.0 - w
	var sb := StyleBoxFlat.new()
	sb.bg_color = Color(0.07, 0.08, 0.09, 0.95)   # solid: chips behind must not show through

	sb.border_color = LINE
	sb.set_border_width_all(1)
	_cv287.draw_style_box(sb, Rect2(p, Vector2(w, h)))
	var hy := p.y + 3.5
	_cv287.draw_rect(Rect2(p.x + 4.0, hy + 3.0, 5.0, 4.0), Color("#5e9a45"))   # HP mark (green block = the bar colour)
	_cv287.draw_string(mf, Vector2(p.x + 11.0, hy + 8.0), hpt, HORIZONTAL_ALIGNMENT_LEFT, -1, 8, INK)
	if arm > 0.0: _chip(Vector2(p.x + 4.0 + top_w - _chip_w("armor", R_num(arm)), hy), "armor", R_num(arm), 1.0, false)

func _attached(v: Dictionary, br: Rect2) -> void:   # s287c (owner): no cards — the HP bar is the card line
	# above the bar = what used to sit above the card (armour / burn / shock numbers, then buff / debuff chips);
	# below the cooldown bar = what used to be inside the card (3 rows x 2 values, left first). Same rules as the cards.
	var a := 1.0
	if int(v["uid"]) == picked or int(v["uid"]) == _hov287: _panel_q.append([v, br])   # s287d: values only on tap / hover
	var row := 0
	var hx := br.position.x
	var base := br.position.y - 1.0
	var st := _row1(v)
	for it in st:
		var pw := _chip_w(it[0], it[1]) + 4.0
		if hx > br.position.x and hx + pw > br.end.x + 8.0:
			row += 1; hx = br.position.x
		var top := base - G - CH - float(row) * (CH + G)
		_chip(Vector2(hx + 2.0, top + 1.5), it[0], it[1], a, true)
		_tips.append([Rect2(hx, top, pw, CH), TIP_ROW1.get(it[0], "") % it[1]])
		hx += pw + G
	var ef := _effects(v)
	if not ef.is_empty():
		if not st.is_empty():
			row += 1; hx = br.position.x
		for e in ef:
			if hx > br.position.x and hx + EF_SZ > br.end.x + 1.0:
				row += 1; hx = br.position.x
			var er := Rect2(hx, base - G - CH - float(row) * (CH + G), EF_SZ, CH)
			_effect_chip(er, e, a)
			_tips.append([er, String(e[4])])
			hx += EF_SZ + G

func _feet_bars(v: Dictionary) -> void:
	if v.get("down") == true or float(v["hp"]) <= 0.0: return
	var g: Variant = _grp(int(v["uid"]))
	if g == null: return
	var sh: Variant = g.get("shadow")
	if not (sh is Node3D) or not (sh as Node3D).is_visible_in_tree(): return
	var cam := get_viewport().get_camera_3d()
	if cam == null: return
	var gp := (sh as Node3D).global_position
	if cam.is_position_behind(gp): return
	var inv := get_global_transform_with_canvas().affine_inverse()
	var c: Vector2 = inv * cam.unproject_position(gp)
	var low: Vector2 = inv * cam.unproject_position(gp + Vector3(0, 0, 0.40))   # front edge of the shadow
	var x0 := roundf(c.x - BAR_W * 0.5)
	var y0 := roundf(low.y + 4.0)
	var mx := maxf(1.0, float(v["maxHp"]))
	var uidb := int(v["uid"])
	_mx0[uidb] = maxf(float(_mx0.get(uidb, 0.0)), mx)
	var cut := 1.0 - mx / float(_mx0[uidb])
	if OS.get_environment("S286FAKESTAT") == "1" and _order.find(uidb) == 0: cut = 0.3   # stills: max HP cut 30 %
	var hp := clampf(float(v["hp"]), 0.0, mx)
	var fr := hp / mx
	# r35h: soft bar — see-through edge, muted colours, 1 px rounded ends (owner: blend with the background)
	var full := HP_H + 3.0
	_bar287[uidb] = Rect2(x0, y0, BAR_W, full)   # s287c: HP + cooldown bar = the divider line
	_soft_rect(Rect2(x0 - 1.0, y0 - 1.0, BAR_W + 2.0, full + 2.0), Color(0.07, 0.06, 0.05, 0.62))
	_soft_rect(Rect2(x0, y0, BAR_W, HP_H), Color(0.09, 0.08, 0.06, 0.65))
	var arm287 := maxf(0.0, float(v.get("armor", 0.0)))
	if OS.get_environment("S286FAKESTAT") == "1" and _order.find(uidb) == 1: arm287 = mx * 0.25   # stills: show the armour part
	var sc287 := BAR_W * (1.0 - cut) / maxf(mx, hp + arm287)   # s287d: armour = white part right after the HP (owner C1)
	var hw := roundf(hp * sc287)
	if hw > 0.0: _soft_rect(Rect2(x0, y0, hw, HP_H), Color("#5e9a45", 0.97) if fr > 0.3 else Color("#bf8530", 0.97))
	var aw287 := roundf((hp + arm287) * sc287) - hw
	if aw287 > 0.0: draw_rect(Rect2(x0 + hw, y0, aw287, HP_H), Color(0.93, 0.93, 0.9, 0.95))
	if hw > 2.0: draw_rect(Rect2(x0 + 1.0, y0, hw - 2.0, 1.0), Color(1.0, 0.98, 0.85, 0.14))   # sun-lit top edge
	if cut > 0.001:   # r35l: max HP cut = black part eating the bar from the right
		var cw := roundf(BAR_W * cut)
		draw_rect(Rect2(x0 + BAR_W - cw, y0, cw, HP_H), Color(0.02, 0.02, 0.02, 0.95))
	var col: Color = Color("#4f7fc4") if int(v["side"]) == 0 else Color("#c0564f")
	var cf := _cd_frac(v)
	draw_rect(Rect2(x0, y0 + HP_H + 1.0, BAR_W, 2.0), Color(0.09, 0.08, 0.06, 0.6))
	if cf >= 0.0:
		var rem_s := _cd_rem(v)
		draw_rect(Rect2(x0, y0 + HP_H + 1.0, BAR_W * cf, 2.0), Color(1, 0.98, 0.9, 0.85) if rem_s < 0.4 else Color(col, 0.97))

func _soft_rect(r: Rect2, c: Color) -> void:   # r35h: rectangle with the 4 corner pixels left out
	if r.size.x < 3.0 or r.size.y < 3.0:
		draw_rect(r, c); return
	draw_rect(Rect2(r.position.x + 1.0, r.position.y, r.size.x - 2.0, r.size.y), c)
	draw_rect(Rect2(r.position.x, r.position.y + 1.0, 1.0, r.size.y - 2.0), c)
	draw_rect(Rect2(r.end.x - 1.0, r.position.y + 1.0, 1.0, r.size.y - 2.0), c)

func _card(r: Rect2, v: Dictionary, on: bool, foe := false) -> void:

	var a := 0.45 if dead else 1.0
	var sb := StyleBoxFlat.new()
	sb.bg_color = PLATE_PICK if on else PLATE
	sb.set_corner_radius_all(3)
	sb.border_color = Color(1, 1, 1, 0.95) if on else LINE
	sb.set_border_width_all(2 if on else 1)
	if OS.get_environment("S293BOX") == "1":
		draw_style_box(sb, r)
	else:   # s293 r35s5: run-end fade, dark at the screen edge, clear past the card's inner edge

static func short_num(x: float) -> String:
	var ax := absf(x)
	var sg := "-" if x < 0.0 else ""
	if ax < 1000.0: return sg + R_raw(ax)
	for u in [[1e9, "B"], [1e6, "M"], [1e3, "K"]]:
		if ax >= float(u[0]):
			var q := ax / float(u[0])
			if q < 9.95:
				var t := "%.1f" % (floorf(q * 10.0) / 10.0)
				if t.ends_with(".0"): t = t.substr(0, t.length() - 2)
				return sg + t + String(u[1])
			return sg + str(int(floorf(q))) + String(u[1])
	return sg + R_raw(ax)

static func R_num(x: float) -> String:
	return short_num(x)

static func R_raw(x: float) -> String:
	return str(int(x)) if absf(x - roundf(x)) < 0.001 else "%.1f" % x

	if txt == "" and _pin_tip != "" and Time.get_ticks_msec() / 1000.0 < _pin_t:
		txt = _pin_tip; at = _pin_r
	if txt == "": return
	var f: Font = TH.font("body")
	var fs := 10
	var tw := f.get_string_size(txt, HORIZONTAL_ALIGNMENT_LEFT, -1, fs).x
	var box := Rect2(clampf(at.position.x, 4.0, size.x - tw - 14.0), at.position.y - 20.0, tw + 10.0, 16.0)
	var sb := StyleBoxFlat.new()
	sb.bg_color = Color("#16202a", 0.94)
	sb.border_color = Color(1, 1, 1, 0.35)
	sb.set_border_width_all(1)
	sb.set_corner_radius_all(3)
	_ic_layer.draw_style_box(sb, box)
	_ic_layer.draw_string(f, box.position + Vector2(5.0, 12.0), txt, HORIZONTAL_ALIGNMENT_LEFT, -1, fs, INK)

func _pix(o: Vector2, g: Array, col: Color, px: float, a: float) -> void:
	for pass_i in 2:
		for yy in g.size():
			var row: String = g[yy]
			for xx in row.length():
				if row[xx] != "#": continue
				if pass_i == 0:
					draw_rect(Rect2(o.x + float(xx) * px - 0.5, o.y + float(yy) * px - 0.5, px + 1.0, px + 1.0), Color(OUT, OUT.a * a))
				else:
					draw_rect(Rect2(o.x + float(xx) * px, o.y + float(yy) * px, px, px), Color(col, a))

func _txt(f: Font, p: Vector2, t: String, fs: int, col: Color, maxw: float) -> void:
	var s := t
	if maxw > 0.0:
		while s.length() > 1 and f.get_string_size(s, HORIZONTAL_ALIGNMENT_LEFT, -1, fs).x > maxw:
			s = s.substr(0, s.length() - 1)
		if s != t: s = s.substr(0, maxi(1, s.length() - 1)) + "…"
	draw_string_outline(f, p, s, HORIZONTAL_ALIGNMENT_LEFT, -1, fs, 3, Color(OUT, OUT.a * col.a))
	draw_string(f, p, s, HORIZONTAL_ALIGNMENT_LEFT, -1, fs, col)

func _cd_rem(v: Dictionary) -> float:
	if not v.has("cdl"): return -1.0
	return maxf(0.0, float(v["cdl"]) - maxf(0.0, float(view.clock) - float(v["cdlT"]))) / 1000.0

func _cd_frac(v: Dictionary) -> float:
	var rem := _cd_rem(v)
	if rem < 0.0: return -1.0
	var st: Variant = v.get("stat")
	var cd := float(st.get("cd", 0.0)) / 1000.0 if st is Dictionary else 0.0
	cd = maxf(cd, rem)
	if cd <= 0.0: return 1.0
	return clampf(1.0 - rem / cd, 0.0, 1.0)

func _stun_left(v: Dictionary) -> float:
	if float(v.get("stunL", 0.0)) <= 0.0: return 0.0
	return maxf(0.0, float(v["stunL"]) - maxf(0.0, float(view.clock) - float(v.get("stunT", 0.0)))) / 1000.0

# ---------- picking: tap a card, or tap a unit (either side) on the arena ----------
func _input(ev: InputEvent) -> void:
	if not is_visible_in_tree() or view == null: return
	var mb := ev as InputEventMouseButton
	if mb == null or not mb.pressed or mb.button_index != MOUSE_BUTTON_LEFT: return
	var p := get_global_transform_with_canvas().affine_inverse() * mb.position
	if picked >= 0 and _det_r.has_point(p):   # s293: tap the detail panel = close it
		_pick(picked)
		get_viewport().set_input_as_handled()
		return
	for i in range(_tips.size() - 1, -1, -1):   # r35l: a chip / a \"+N\" card tap shows its detail
		var tr: Rect2 = _tips[i][0]
		if OS.get_environment("S293TIPS") == "1" and tr.has_point(p) and not _rects.values().has(tr):
			_pin_tip = _tips[i][1]; _pin_r = tr; _pin_t = Time.get_ticks_msec() / 1000.0 + 2.5
			get_viewport().set_input_as_handled()
			return
	for uid in _rects:
		if (_rects[uid] as Rect2).grow_individual(0, 26, 0, 26).has_point(p):   # s293: card + its chip rows
			_pick(uid)
			get_viewport().set_input_as_handled()
			return
	var hit := _unit_at(mb.position)
	if hit >= 0:
		_pick(hit)

func _pick(uid: int) -> void:
	picked = -1 if picked == uid else uid
	_det_t0 = Time.get_ticks_msec() / 1000.0   # s293 r35s3: open animation restarts on every pick
	if OS.get_environment("S286LOG") == "1": print("S286 pick ", picked)

func _unit_at(screen_p: Vector2) -> int:
	var cam := get_viewport().get_camera_3d()
	if cam == null: return -1
	var canvas_p := get_viewport().get_canvas_transform().affine_inverse() * screen_p
	var best := -1
	var bestd := 1.0e9
	for v in view.units.values():
		if v.get("down") == true or float(v["hp"]) <= 0.0: continue
		var uid := int(v["uid"])
		var s := _spr(uid)
		if s == null or not s.is_visible_in_tree(): continue
		var aabb := s.get_aabb()
		var rr := Rect2(cam.unproject_position(s.global_transform * aabb.get_endpoint(0)), Vector2.ZERO)
		for i in 8:
			rr = rr.expand(cam.unproject_position(s.global_transform * aabb.get_endpoint(i)))
		if rr.grow(-rr.size.x * 0.15).has_point(canvas_p):
			var d := canvas_p.distance_to(rr.get_center())
			if d < bestd:
				bestd = d; best = uid
	return best

# ---------- white outline on the picked unit's sprite: 4 white silhouettes 1 art pixel out, just behind it ----------
func _silhouette(t: Texture2D) -> Texture2D:
	if _sil.has(t): return _sil[t]
	var img := t.get_image()
	if img == null: return null
	img = img.duplicate()
	img.decompress()
	img.convert(Image.FORMAT_RGBA8)
	for yy in img.get_height():
		for xx in img.get_width():
			img.set_pixel(xx, yy, Color(1, 1, 1, 1) if img.get_pixel(xx, yy).a > 0.5 else Color(1, 1, 1, 0))
	var out := ImageTexture.create_from_image(img)
	_sil[t] = out
	return out

func _clear_outline() -> void:
	for o in _outl:
		if is_instance_valid(o): o.queue_free()
	_outl.clear()
	_outl_src = null

func _sync_outline() -> void:
	var s := _spr(picked) if picked >= 0 else null
	if s != null and view.units.has(picked) and (view.units[picked].get("down") == true or float(view.units[picked]["hp"]) <= 0.0):
		s = null
	if s == null:
		if _outl_src != null: _clear_outline()
		return
	if s != _outl_src or (not _outl.is_empty() and _outl[0].get_meta("src") != s.texture):
		_clear_outline()
		var sil := _silhouette(s.texture)
		if sil == null: return
		_outl_src = s
		for d in [Vector2(1, 0), Vector2(-1, 0), Vector2(0, 1), Vector2(0, -1)]:
			var o := Sprite3D.new()
			o.texture = sil
			o.texture_filter = BaseMaterial3D.TEXTURE_FILTER_NEAREST
			o.alpha_cut = SpriteBase3D.ALPHA_CUT_DISCARD
			o.shaded = false
			o.double_sided = true
			o.cast_shadow = GeometryInstance3D.SHADOW_CASTING_SETTING_OFF
			o.set_meta("d", d)
			o.set_meta("src", s.texture)
			s.add_child(o)
			_outl.append(o)
	for o in _outl:
		var d: Vector2 = o.get_meta("d")
		o.hframes = s.hframes; o.vframes = s.vframes; o.frame = s.frame
		o.flip_h = s.flip_h; o.flip_v = s.flip_v
		o.centered = s.centered; o.offset = s.offset
		o.pixel_size = s.pixel_size
		o.region_enabled = s.region_enabled; o.region_rect = s.region_rect
		o.render_priority = s.render_priority - 1
		o.position = Vector3(d.x * s.pixel_size, d.y * s.pixel_size, -0.004)
