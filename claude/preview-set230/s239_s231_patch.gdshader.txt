// s231 FF Re-style grass patch (own code, numbers from hd-2d-ref): small leaning blade cards, normals = straight up
// (lit exactly like the ground), vertex colour R 0 root -> 1 tip (root darkening + wind mask), baked shade by world xz.
shader_type spatial;
render_mode cull_disabled, depth_draw_opaque;
uniform sampler2D tex : source_color, filter_linear_mipmap_anisotropic;
uniform vec3 tint = vec3(1.0);
uniform float bright = 1.0;
uniform float root_dark = 0.55;
uniform float wind = 0.04;
uniform float wind_speed = 0.9;
uniform bool use_shade = false;
uniform sampler2D shade_tex : filter_linear_mipmap;
uniform float shade_ext = 26.0;
uniform float gt = 0.0;          // v36: shared gust clock (same as the backdrop card / petals)
uniform float gust_period = 15.0;
uniform float gust_cross = 1.6;   // v38: seconds for the gust to sweep right -> left
uniform float gust_w = 0.07;       // v38: swell width (share of the period)
uniform float gust_lean = 0.0;   // v36: lean to screen-left at the gust peak (0 = off)
uniform float gust_amp = 0.0;    // v36: extra flutter at the gust peak
uniform float fg_sat = 1.0;      // v36: tone toward the backdrop
uniform vec3 fg_tone = vec3(1.0);
float gust_at(float x) { float ph = fract((gt + x * gust_cross) / gust_period); return exp(-((ph - 0.25) / gust_w) * ((ph - 0.25) / gust_w)); }
vec3 fg_grade(vec3 c) { return mix(vec3(dot(c, vec3(0.299, 0.587, 0.114))), c, fg_sat) * fg_tone; }
varying vec3 wp;
void vertex() {
	wp = (MODEL_MATRIX * vec4(VERTEX, 1.0)).xyz;
	float w = sin(TIME * wind_speed + wp.x * 0.35 + wp.z * 0.21) * 0.6 + sin(TIME * wind_speed * 2.3 + wp.x * 0.9) * 0.4;
	vec4 cp = PROJECTION_MATRIX * (MODELVIEW_MATRIX * vec4(VERTEX, 1.0));
	float big = gust_at(clamp(cp.x / cp.w * 0.5 + 0.5, 0.0, 1.0));
	VERTEX.xz += vec2(1.0, 0.3) * w * wind * (1.0 + gust_amp * big) * COLOR.r * COLOR.r;
	vec3 lw = -normalize(vec3(INV_VIEW_MATRIX[0].x, 0.0, INV_VIEW_MATRIX[0].z)) * gust_lean * big * COLOR.r * COLOR.r;   // v36: lean with the gust (screen-left, same way the petals fly)
	VERTEX += (inverse(MODEL_MATRIX) * vec4(lw, 0.0)).xyz;
}
void fragment() {
	vec4 c = texture(tex, UV);
	ALPHA = c.a;
	ALPHA_SCISSOR_THRESHOLD = 0.5;
	vec3 col = c.rgb * tint * bright * mix(root_dark, 1.0, COLOR.r);
	if (use_shade) { col *= texture(shade_tex, (wp.xz + shade_ext) / (2.0 * shade_ext)).rgb; }
	ALBEDO = fg_grade(col);
	NORMAL = normalize((VIEW_MATRIX * vec4(0.0, 1.0, 0.0, 0.0)).xyz); // both faces lit like the ground (cull_disabled flips back faces)
	ROUGHNESS = 1.0;
	SPECULAR = 0.0;
}
