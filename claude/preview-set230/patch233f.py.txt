import sys, os
T = sys.argv[1]
p = os.path.join(T, 's226.gd')
s = open(p, encoding='utf-8').read()
a = '\tif EV.has("rfront"): _s233_front(st, EV)'
assert s.count(a) == 1
s = s.replace(a, a + '\n\tif EV.has("wl"): _s233_water(st, EV) # s233 R13: sea in the valley (owner W1)')
s += '''

# s233 R13 (owner W1): one flat sea sheet in the valley, like FF's water001 (a flat mesh below the field) [DATA];
# the look (sky tint by angle, slow ripples, light shoreline) is ours [APPROX]. The shore needs the ground height, so the
# water shader reuses the terrain shader's own uniforms + th() (taken from its code: one formula, no copy to drift).
const WATER233 := """
uniform float wl = -22.0;
uniform vec3 deep = vec3(0.16, 0.36, 0.46);
uniform vec3 skyc = vec3(0.66, 0.80, 0.90);
uniform vec3 shore_c = vec3(0.78, 0.86, 0.80);
uniform float shore_w = 2.2;
varying vec3 wp2;
void vertex() { wp2 = (MODEL_MATRIX * vec4(VERTEX, 1.0)).xyz; }
void fragment() {
	float g = th(wp2.xz);
	float depth = wl - g;
	if (depth < -0.05) { discard; }
	float fres = pow(1.0 - clamp(dot(NORMAL, VIEW), 0.0, 1.0), 2.2);
	vec2 q = wp2.xz * 0.08;
	float rip = vn(q + vec2(TIME * 0.05, TIME * 0.03)) * 0.6 + vn(q * 3.1 - vec2(TIME * 0.08, 0.0)) * 0.4;
	vec3 c = mix(deep, skyc, clamp(fres + (rip - 0.5) * 0.25, 0.0, 1.0));
	c += vec3(0.9, 0.85, 0.7) * smoothstep(0.82, 0.95, rip) * 0.12; // sparkles
	float sh = 1.0 - smoothstep(0.0, shore_w, depth);
	c = mix(c, shore_c, sh * 0.75);
	ALBEDO = c;
}
"""
func _s233_water(st: Node, EV: Dictionary) -> void:
	var old := st.find_child("Ground", true, false) as MeshInstance3D
	var tcode := (load("res://tests/s233_terrain.gdshader") as Shader).code
	var cut := tcode.find("void vertex()")
	var code := tcode.substr(0, cut).replace("render_mode specular_disabled;", "render_mode unshaded, cull_disabled;") + WATER233
	var sh := Shader.new()
	sh.code = code
	var m := _s233_mat(sh, old.material_override as ShaderMaterial, 0.0, 0.0, EV)
	var wl := float(EV["wl"])
	m.set_shader_parameter("wl", wl)
	for k in ["deep", "skyc", "shore_c"]:
		if EV.has("w" + k):
			var cc := Color(EV["w" + k])
			m.set_shader_parameter(k, Vector3(cc.r, cc.g, cc.b))
	var w := MeshInstance3D.new()
	w.name = "S233Sea"
	var pm := PlaneMesh.new()
	pm.size = Vector2(800, 520)
	w.mesh = pm
	w.material_override = m
	w.cast_shadow = GeometryInstance3D.SHADOW_CASTING_SETTING_OFF
	st.add_child(w)
	w.global_position = Vector3(0, wl, -280)
	print("s233 sea at ", wl)
'''
open(p, 'w', encoding='utf-8').write(s)
print('patched')
