import sys, os
T = sys.argv[1]
def patch(p, pairs, tail=''):
    p = os.path.join(T, p)
    s = open(p, encoding='utf-8').read()
    for a, b in pairs:
        assert s.count(a) == 1, (p, a[:70])
        s = s.replace(a, b)
    s += tail
    open(p, 'w', encoding='utf-8').write(s)

# s233 B1: rolling ground - same th() as tests/s233_terrain.gdshader (change both). A fine near mesh over the arena +
# the old far Ground gets the same shader, sunk a little so the near mesh wins where both exist.
func _s233_h(p: Vector2) -> float:
	return fposmod(sin(p.dot(Vector2(127.1, 311.7))) * 43758.5453, 1.0)
func _s233_vn(p: Vector2) -> float:
	var i := p.floor()
	var f := p - i
	f = f * f * (Vector2(3, 3) - 2.0 * f)
	return lerpf(lerpf(_s233_h(i), _s233_h(i + Vector2(1, 0)), f.x), lerpf(_s233_h(i + Vector2(0, 1)), _s233_h(i + Vector2(1, 1)), f.x), f.y)
func _s233_fbm(p: Vector2) -> float:
	return _s233_vn(p) * 0.55 + _s233_vn(p * 2.03 + Vector2(5.1, 5.1)) * 0.3 + _s233_vn(p * 4.1 + Vector2(9.7, 9.7)) * 0.15
func _s233_th(p: Vector2, EV: Dictionary) -> float:
	var r := p.length() / 11.7
	var shelf := -0.04 - 0.11 * smoothstep(1.15, 1.6, r)
	var mound := float(EV.get("tamp", "1.6")) * smoothstep(1.35, 2.8, r) * pow(_s233_fbm(p * 0.07), 1.3)
	var dr := Vector2(p.x * 0.85, p.y + 2.0).length()
	return shelf + mound - 60.0 * smoothstep(20.0, 45.0, dr)

func _s233_mat(sh: Shader, om: ShaderMaterial, node_y: float, sink: float, EV: Dictionary) -> ShaderMaterial:
	var m := ShaderMaterial.new()
	m.shader = sh
	m.set_shader_parameter("tex", om.get_shader_parameter("tex"))
	m.set_shader_parameter("tint", om.get_shader_parameter("tint"))
	m.set_shader_parameter("tile", om.get_shader_parameter("tile"))
	m.set_shader_parameter("teal", float(EV.get("teal", "0.22")))
	m.set_shader_parameter("amp", float(EV.get("tamp", "1.6")))
	m.set_shader_parameter("node_y", node_y)
	m.set_shader_parameter("sink", sink)
	m.set_shader_parameter("dirt_amt", float(EV.get("tdirt", "1.0")))
	m.set_shader_parameter("dry_amt", float(EV.get("tdry", "0.5")))
	m.set_shader_parameter("path_amt", float(EV.get("tpath", "1.0")))
	m.set_shader_parameter("hue_amt", float(EV.get("thue", "1.0")))
	return m

func _s233_terrain(st: Node, EV: Dictionary) -> void:
	var old := st.find_child("Ground", true, false) as MeshInstance3D
	if old == null:
		print("s233 terrain: no Ground")
		return
	var om := old.material_override as ShaderMaterial
	var sh := load("res://tests/s233_terrain.gdshader") as Shader
	old.material_override = _s233_mat(sh, om, old.global_position.y, 0.08, EV)
	old.extra_cull_margin = 80.0
	var near := MeshInstance3D.new()
	near.name = "S233Near"
	var pm := PlaneMesh.new()
	pm.size = Vector2(110, 90)
	pm.subdivide_width = 274
	pm.subdivide_depth = 224
	near.mesh = pm
	st.add_child(near)
	near.global_position = Vector3(0, 0, -25)
	near.material_override = _s233_mat(sh, om, 0.0, 0.0, EV)
	near.extra_cull_margin = 80.0
	print("s233 terrain: near mesh + far ground, amp ", EV.get("tamp", "1.6"))
''')
