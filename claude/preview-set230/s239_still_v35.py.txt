# v35: "still" mask = rocks (grey, low saturation) under the tree band -> no sway/shimmer there
import sys, numpy as np, cv2
c = cv2.imread('s239cut/one_v19/one_nosky_v30.png', cv2.IMREAD_UNCHANGED)
m = cv2.imread('s239cut/one_v19/tree_mask_v32.png', cv2.IMREAD_UNCHANGED)
b = c[:, :, :3]; hsv = cv2.cvtColor(b, cv2.COLOR_BGR2HSV)
B, G, R = [b[:, :, i].astype(np.int16) for i in range(3)]
grey = ((G - np.maximum(R, B)) < 6) & (c[:, :, 3] > 200)   # not green
band = np.zeros_like(grey)
for x0, y0, x1, y1 in [(686, 358, 795, 452), (765, 393, 835, 432), (810, 432, 920, 476), (886, 418, 918, 460), (926, 448, 980, 496)]:   # owner's rock group (picture px)
    band[y0:y1, x0:x1] = True
rock = (grey & band).astype(np.uint8)
rock = cv2.morphologyEx(rock, cv2.MORPH_OPEN, np.ones((3, 3), np.uint8))
n, lab, st, _ = cv2.connectedComponentsWithStats(rock)
keep = np.zeros_like(rock)
for i in range(1, n):
    if st[i, 4] >= 25: keep[lab == i] = 1
keep = cv2.morphologyEx(keep, cv2.MORPH_CLOSE, np.ones((7, 7), np.uint8))
halo = cv2.dilate(keep, cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (25, 25)))
still = cv2.GaussianBlur(halo.astype(np.float32), (0, 0), 4.0); still = np.maximum(still, keep)
a = (255 * (1 - np.clip(still, 0, 1))).astype(np.uint8)
cv2.imwrite('s239cut/one_v19/tree_mask_v35.png', np.dstack([m, a]))
o = b.copy(); o[keep > 0] = (255, 0, 255); o[(halo > 0) & (keep == 0)] = o[(halo > 0) & (keep == 0)] // 2 + (0, 60, 60)
cv2.imwrite('out/s239v35/still_check.png', o[340:500, 640:1000])
print('rock px', int(keep.sum()), 'blobs', n - 1)
