// set 239 sample (own code): the background picture card, brought to life without video.
// mask R = tree foliage, G = lake water, B = how high in the crown (tops sway more, trunks/bases stay put)
shader_type spatial;
render_mode unshaded, cull_disabled, fog_disabled, shadows_disabled;
uniform sampler2D tex : source_color, filter_linear;
uniform sampler2D mask : filter_linear;
uniform float t = 0.0;
uniform float sway = 0.0011;   // max UV shift of a crown top (~2 px on the 2000 px picture)
uniform float sparkle = 0.85;
float h(vec2 p) { return fract(sin(dot(p, vec2(127.1, 311.7))) * 43758.5453); }
float vn(vec2 p) {
	vec2 i = floor(p), f = fract(p);
	vec2 u = f * f * (3.0 - 2.0 * f);
	return mix(mix(h(i), h(i + vec2(1, 0)), u.x), mix(h(i + vec2(0, 1)), h(i + vec2(1, 1)), u.x), u.y);
}
void fragment() {
	vec3 m = texture(mask, UV).rgb;
	// wind: a slow gust that travels left->right across the picture, plus small leaf flutter
	float gust = 0.6 + 0.4 * sin(t * 0.7 - UV.x * 6.0) ;
	float wob = sin(t * 1.6 + UV.x * 55.0 + UV.y * 30.0) * 0.6 + (vn(UV * vec2(90.0, 60.0) + vec2(t * 0.8, 0.0)) - 0.5) * 0.8;
	float amt = m.r * (0.25 + 0.75 * m.b) * gust;
	vec2 uv = UV + vec2(wob, wob * 0.35) * sway * amt;
	vec4 c = texture(tex, uv);
	// lake: tiny glints that wink on and off as small waves pass
	float w = vn(UV * vec2(900.0, 2600.0) + vec2(t * 1.5, 0.0)) * vn(UV * vec2(500.0, 1500.0) - vec2(t * 0.9, 0.0));
	float glint = smoothstep(0.42, 0.62, w) * m.g * sparkle;
	ALBEDO = c.rgb + vec3(1.0, 0.98, 0.92) * glint;
	ALPHA = texture(tex, UV).a;
	ALPHA_SCISSOR_THRESHOLD = 0.5;
}
