// set 239 sample (own code): drifting mist band in front of the background card - two octaves of value noise scrolled by t
shader_type spatial;
render_mode unshaded, cull_disabled, fog_disabled, shadows_disabled, depth_draw_never;
uniform vec3 col : source_color = vec3(0.92, 0.94, 0.95);
uniform float amt = 0.45;
uniform float t = 0.0;
uniform float speed = 0.012;
uniform vec2 scale = vec2(6.0, 2.2);
float h(vec2 p) { return fract(sin(dot(p, vec2(127.1, 311.7))) * 43758.5453); }
float vn(vec2 p) {
	vec2 i = floor(p), f = fract(p);
	vec2 u = f * f * (3.0 - 2.0 * f);
	return mix(mix(h(i), h(i + vec2(1, 0)), u.x), mix(h(i + vec2(0, 1)), h(i + vec2(1, 1)), u.x), u.y);
}
float fbm(vec2 p) { float s = 0.0, a = 0.5; for (int k = 0; k < 5; k++) { s += a * vn(p); p = p * 2.03 + 17.0; a *= 0.5; } return s; }
void fragment() {
	vec2 p = UV * scale + vec2(t * speed * scale.x, 0.0);
	float n = fbm(p + vec2(fbm(p * 0.7 - vec2(t * speed * 3.0, 0.0)) * 1.2, 0.0)); // domain warp = slow rolling wisps
	float band = smoothstep(0.0, 0.35, UV.y) * smoothstep(1.0, 0.55, UV.y);
	float ends = smoothstep(0.0, 0.15, UV.x) * smoothstep(1.0, 0.85, UV.x);
	ALBEDO = col;
	ALPHA = clamp((n - 0.38) * 2.2, 0.0, 1.0) * band * ends * amt;
}
