// set 239 v33 sample (own code): a tiny butterfly over the bushes. Billboard; wings flap by squashing their width.
shader_type spatial;
render_mode unshaded, cull_disabled, fog_disabled, shadows_disabled, blend_mix, depth_draw_never;
uniform vec3 col = vec3(1.0, 0.95, 0.75);
uniform float flap = 1.0;   // 0..1 wing width
uniform float tilt = 0.0;   // rad, banks the body along its path
uniform float alpha = 1.0;
void vertex() {
	float c = cos(tilt), s = sin(tilt);
	vec2 v = vec2(c * VERTEX.x - s * VERTEX.y, s * VERTEX.x + c * VERTEX.y);
	MODELVIEW_MATRIX = VIEW_MATRIX * mat4(INV_VIEW_MATRIX[0], INV_VIEW_MATRIX[1], INV_VIEW_MATRIX[2], MODEL_MATRIX[3]);
	VERTEX = vec3(v, 0.0);
}
void fragment() {
	vec2 p = UV * 2.0 - 1.0;
	float w = max(flap, 0.12);
	vec2 q = vec2(abs(p.x) / w, p.y);
	float up = 1.0 - smoothstep(0.75, 0.95, length((q - vec2(0.5, -0.28)) / vec2(0.5, 0.48)));   // fore wing
	float lo = 1.0 - smoothstep(0.75, 0.95, length((q - vec2(0.38, 0.38)) / vec2(0.36, 0.34)));  // hind wing
	float body = (1.0 - smoothstep(0.06, 0.12, abs(p.x))) * (1.0 - smoothstep(0.55, 0.7, abs(p.y)));
	float a = max(max(up, lo) * step(abs(p.x), w), body);
	vec3 c = mix(col, col * 0.7, smoothstep(0.2, 0.9, length(q - vec2(0.45, -0.1))));
	c = mix(c, vec3(0.18, 0.15, 0.12), body);
	ALBEDO = c;
	ALPHA = a * alpha;
}
