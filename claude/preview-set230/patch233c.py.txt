import sys, os
T = sys.argv[1]
def patch(p, pairs, tail=''):
    p = os.path.join(T, p)
    s = open(p, encoding='utf-8').read()
    for a, b in pairs:
        assert s.count(a) == 1, (p, a[:70])
        s = s.replace(a, b)
    s += tail
    open(p, 'w', encoding='utf-8').write(s)
