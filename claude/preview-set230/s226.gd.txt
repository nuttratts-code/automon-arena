func _card(st: Node, file: String, at: Vector3, width: float, tint: Color) -> void:
	var t: Variant = load(file)
	if not (t is Texture2D):
		return
	var sp := Sprite3D.new()
	sp.texture = t
	sp.pixel_size = width / float((t as Texture2D).get_width())
	sp.billboard = BaseMaterial3D.BILLBOARD_FIXED_Y
	sp.alpha_cut = SpriteBase3D.ALPHA_CUT_DISCARD
	sp.shaded = false
	sp.modulate = tint
	sp.cast_shadow = GeometryInstance3D.SHADOW_CASTING_SETTING_OFF
	sp.position = at + Vector3(0, width * float((t as Texture2D).get_height()) / float((t as Texture2D).get_width()) * 0.5 - 0.12, 0)
	st.add_child(sp)

func _b6(st: Node, cam: Camera3D, EV: Dictionary) -> void:
	# 1 ground: shift the yellow-green toward the ref's blue-green (74,127,111) — a tint cannot add blue the texture lacks
	var teal := float(EV.get("teal", "0.5"))
	for mi in st.find_children("*", "MeshInstance3D", true, false):
		var mo: Material = (mi as MeshInstance3D).material_override
		if mo is ShaderMaterial and (mo as ShaderMaterial).get_shader_parameter("tint") != null:
			var sh := Shader.new()
			sh.code = (mo as ShaderMaterial).shader.code.replace("ALBEDO = c * tint;", "float l = dot(c, vec3(0.3, 0.59, 0.11));
	c = mix(c, l * vec3(0.78, 1.32, 1.18), %s);
	ALBEDO = c * tint;" % str(teal))
			(mo as ShaderMaterial).shader = sh
	# 2 plaza: warm beige like the ref's dirt path, a bit darker (owner tone script on the Plaza node)
	for n in st.find_children("Plaza", "", true, false):
		if "tone" in n:
			var pt: Array = str(EV.get("plaza", "0.66,0.6,0.48")).split_floats(",")
			n.tone = Color(pt[0], pt[1], pt[2])
	# 3 grass tufts: a ring at the plaza rim (back + sides), never inside the board
	var rng := RandomNumberGenerator.new()
	rng.seed = 22501
	var files := ["grass0", "grass1", "grass2", "grass3", "grass4", "grass5", "grass_big"]
	var gtint := Color(0.62, 0.74, 0.66)
	var ring_n := int(EV.get("tufts", "70"))
	for i in ring_n:
		var a := rng.randf_range(PI * 0.92, PI * 2.08) # back half + the two sides
		var r := rng.randf_range(15.0, 21.0)
		var at := Vector3(cos(a) * r * 1.12, 0, sin(a) * r - 1.5)
		var w := rng.randf_range(float(EV.get("tw0", "0.7")), float(EV.get("tw1", "1.4")))
		_card(st, "res://art/s217/%s.png" % files[rng.randi() % files.size()], at, w, gtint * rng.randf_range(0.85, 1.05))
	# 4 foreground framing: big tufts at the bottom corners, soft in the tilt-shift band (ref: grass between camera and field)
	if EV.get("fg", "1") == "1": # the view's bottom edge meets the ground at z ~7, half-width ~6 m there
		for fx in [[-7.4, 7.6, 2.2], [-5.6, 8.4, 1.6], [7.4, 7.8, 2.1], [5.8, 8.6, 1.5]]:

# s229: the plaza looked like it floats — no cast shadows in Compatibility, a clean cut edge, bright even grass.
# Fake contact shadow + darker grass band (one flat ring, unlit, alpha) + small tufts sitting ON the rim.
func _rim(st: Node, EV: Dictionary) -> void:
	var box := AABB()
	var first := true
	for n in st.find_children("Plaza", "", true, false):
		for mi in n.find_children("*", "MeshInstance3D", true, false):
			var b: AABB = (mi as MeshInstance3D).global_transform * (mi as MeshInstance3D).get_aabb()
			box = b if first else box.merge(b)
			first = false
	if first:
		return
	var c := box.get_center()
	var r := minf(box.size.x, box.size.z) * 0.5 * float(EV.get("rimr", "0.97"))
	var top := box.end.y
	var outer := r * float(EV.get("rimw", "1.45"))
	print("s229 plaza centre ", c, " r ", r, " top ", top, " bottom ", box.position.y)
	var mi := MeshInstance3D.new()
	var pm := PlaneMesh.new()
	pm.size = Vector2(outer * 2.0, outer * 2.0)
	mi.mesh = pm
	var sh := Shader.new()
	sh.code = """shader_type spatial;
render_mode unshaded, blend_mix, depth_draw_never, cull_disabled;
uniform float inner; uniform float a_edge; uniform float a_band; uniform vec3 col;
void fragment() {
	float d = length(UV - 0.5) * 2.0;
	float edge = 1.0 - smoothstep(inner, inner + 0.07, d);
	float band = 1.0 - smoothstep(inner, 1.0, d);
	float a = max(a_edge * edge, a_band * band) * step(inner * 0.985, d);
	ALBEDO = col; ALPHA = a;
}"""
	var m := ShaderMaterial.new()
	m.shader = sh
	m.set_shader_parameter("inner", r / outer)
	m.set_shader_parameter("a_edge", float(EV.get("rima", "0.6")))
	m.set_shader_parameter("a_band", float(EV.get("rimb", "0.3")))
	m.set_shader_parameter("col", Vector3(1, 0, 0) if EV.get("rimdbg", "0") == "1" else Vector3(0.05, 0.07, 0.04))
	mi.material_override = m
	mi.cast_shadow = GeometryInstance3D.SHADOW_CASTING_SETTING_OFF
	mi.position = Vector3(c.x, float(EV.get("rimy", "0.02")), c.z)
	st.add_child(mi)
	# small tufts on the rim, back half + sides (the front rim is under the foreground grass strip)
	var rng := RandomNumberGenerator.new()
	rng.seed = 22901
	var files := ["grass0", "grass1", "grass2", "grass3", "grass4", "grass5"]
	for i in int(EV.get("rimt", "90")):
		var a := rng.randf_range(PI * 0.9, PI * 2.1)
		var rr := r * rng.randf_range(0.95, 1.06)
		_card(st, "res://art/s217/%s.png" % files[rng.randi() % files.size()], Vector3(c.x + cos(a) * rr, 0.0, c.z + sin(a) * rr), rng.randf_range(0.35, 0.75), Color(0.5, 0.62, 0.52) * rng.randf_range(0.85, 1.05))

# s230 "baked" shade (FF Re bakes lightmaps; Compatibility has no live sun shadows): once per shot, rasterise every

func _s230_blur(im: Image, r: float) -> Image:
	if r <= 0.0:
		return im
	var w := im.get_width()
	var small := maxi(8, int(w / maxf(r, 1.0)))
	var b := im.duplicate() as Image
	b.resize(small, small, Image.INTERPOLATE_BILINEAR)
	b.resize(w, w, Image.INTERPOLATE_CUBIC)
	return b

# s232 step 4c ①: contact shadow sized to the sprite's own feet — dark core, short hard edge (not the soft round blob).
# Flyers (--e:fly=XA:0.45,…) are lifted; their shadow stays on the ground, smaller and lighter.
# Args: fsa alpha · fsc colour · fsw width x foot span · fsd depth/width · fse edge (0 hard … 1 soft) · fsx shift right (sun from left)
func _s232_foot(s: Sprite3D, sh: MeshInstance3D, img: Image, cid: String, EV: Dictionary) -> void:
	var w := img.get_width(); var h := img.get_height()
	var bottom := 0
	for y in range(h - 1, -1, -1):
		for x in w:
			if img.get_pixel(x, y).a > 0.5: bottom = y; break
		if bottom > 0: break
	var x0 := w; var x1 := 0 # foot span = opaque columns in the lowest fsb (30 %) of the body, centre = their mean
	var sx := 0.0; var nn := 0.0
	for y in range(maxi(0, bottom - int(h * float(EV.get("fsb", "0.3")))), bottom + 1):
		for x in w:
			if img.get_pixel(x, y).a > 0.5:
				x0 = mini(x0, x); x1 = maxi(x1, x); sx += x; nn += 1.0
	var span := float(maxi(x1 - x0, int(w * 0.3))) * s.pixel_size
	var cx := (sx / maxf(nn, 1.0) - float(w) * 0.5) * s.pixel_size * (-1.0 if s.flip_h else 1.0)
	var fly := 0.0
	for e in str(EV.get("fly", "")).split(",", false):
		var kv := e.split(":")
		if kv[0] == cid: fly = float(kv[1])
	var a := float(EV.get("fsa", "0.8")) * (1.0 / (1.0 + fly * 0.9))
	var edge := clampf(float(EV.get("fse", "0.25")) + fly * 0.5, 0.02, 1.0)
	var c := Color(EV.get("fsc", "#10141c"))
	var g := Gradient.new()
	g.offsets = PackedFloat32Array([0.0, 1.0 - edge, 1.0])
	g.colors = PackedColorArray([Color(c, a), Color(c, a * 0.85), Color(c, 0.0)])
	var t := GradientTexture2D.new()
	t.gradient = g; t.fill = GradientTexture2D.FILL_RADIAL
	t.fill_from = Vector2(0.5, 0.5); t.fill_to = Vector2(1.0, 0.5); t.width = 128; t.height = 128
	(sh.material_override as StandardMaterial3D).albedo_texture = t
	var pm := PlaneMesh.new()
	var wd := span * float(EV.get("fsw", "0.9")) * (1.0 / (1.0 + fly * 0.4))
	pm.size = Vector2(wd, wd * float(EV.get("fsd", "0.32")))
	sh.mesh = pm
	sh.scale = Vector3.ONE
	sh.position.x = s.position.x + cx + float(EV.get("fsx", "0.04"))
	if fly > 0.0:
		s.position.y += fly
	if EV.get("fsdbg", "0") == "1": (sh.material_override as StandardMaterial3D).albedo_color = Color(1, 0, 0)

# s232 step 4c ②: sun match on the picture itself — sun from screen-left (light side brighter, far side + feet darker,
# thin warm rim on the lit edge) and loud colours pulled down (soft cap on saturation, so the red ogre stops popping).
# Args: smsat global sat · smcap sat above this is squeezed · smk squeeze · smdir left/right light · smao feet dark · smrim rim
func _s232_sun(img: Image, flipped: bool, EV: Dictionary) -> Image:
	var w := img.get_width(); var h := img.get_height()
	var gs := float(EV.get("smsat", "0.92")); var cap := float(EV.get("smcap", "0.35")); var sk := float(EV.get("smk", "0.45"))
	var dr := float(EV.get("smdir", "0.16")); var ao := float(EV.get("smao", "0.22")); var rim := float(EV.get("smrim", "0.25"))
	var rc := Color(EV.get("smrc", "#ffe2b0"))
	var rw := int(EV.get("smrw", "3"))
	var mc := Color(EV.get("smmc", "#8a8a78")); var mix := float(EV.get("smmix", "0.0"))
	var out := img.duplicate() as Image
	for y in h:
		var v := float(y) / float(h)
		for x in w:
			var c := img.get_pixel(x, y)
			if c.a <= 0.0: continue
			var su := float(x) / float(w)
			if flipped: su = 1.0 - su # screen-left fraction (0 = the sun side)
			var l := c.r * 0.3 + c.g * 0.59 + c.b * 0.11
			var sat := maxf(c.r, maxf(c.g, c.b)) - minf(c.r, minf(c.g, c.b))
			var f := gs
			if sat > cap: f *= (cap + (sat - cap) * sk) / sat
			var r := Color(lerpf(l, c.r, f), lerpf(l, c.g, f), lerpf(l, c.b, f), c.a)
			r = r.lerp(Color(mc.r * l * 2.0, mc.g * l * 2.0, mc.b * l * 2.0, c.a), mix) # bleed the floor colour in
			var m := (1.0 + dr * (0.5 - su) * 2.0) * (1.0 - ao * smoothstep(0.7, 1.0, v))
			r = Color(r.r * m, r.g * m, r.b * m, c.a)
			# rim: transparent pixel within rw px toward the sun side = lit edge
			var sx := -1 if not flipped else 1
			var edge := false
			for d in range(1, rw + 1):
				var xx := x + sx * d
				if xx < 0 or xx >= w or img.get_pixel(xx, y).a < 0.5:
					edge = true; break
			if edge and v < 0.85:
				r = r.lerp(Color(rc.r * maxf(l, 0.6), rc.g * maxf(l, 0.6), rc.b * maxf(l, 0.6), c.a) * 1.25, rim)
				r.a = c.a
			out.set_pixel(x, y, r)
	return out


# s233: motion sample — the scene as it would play: grass wind (shader TIME), pollen, units breathe, the flyer hovers,
# one unit at a time lunges at a foe, the foe flashes + shakes. Run with --fixed-fps so time is even. Own code.
func _s233_anim(st: Node, sprs: Array, map: Dictionary, map_side: Dictionary, nf: int, out: String, EV: Dictionary) -> void:
	var fps := float(EV.get("fps", "30"))
	var ids: Array = map.keys()
	var base := {}
	var lean: Vector3 = st.lean()
	for id in ids:
		var s: Sprite3D = st.groups["battle"][id]["spr"]
		var sh: MeshInstance3D = st.groups["battle"][id]["shadow"]
		base[id] = [s.position, sh.position, s.modulate, sh.scale, s.pixel_size * float(s.texture.get_height()) * 0.5]
	var fly := {}
	for e in str(EV.get("fly", "")).split(",", false):
		fly[e.split(":")[0]] = true
	var beat := float(EV.get("beat", "0.9")) # seconds between attacks
	var rng := RandomNumberGenerator.new()
	rng.seed = 23301
	var plan := [] # [start time, attacker, target]
	var t0 := 0.4
	var side := 0
	while t0 < nf / fps:
		var a: int = ids.filter(func(x): return map_side[x] == side)[rng.randi() % 6]
		var b: int = ids.filter(func(x): return map_side[x] != side)[rng.randi() % 6]
		plan.append([t0, a, b])
		t0 += beat
		side = 1 - side
	for f in nf:
		var t := float(f) / fps
		for i in ids.size():
			var id: int = ids[i]
			var s: Sprite3D = st.groups["battle"][id]["spr"]
			var sh: MeshInstance3D = st.groups["battle"][id]["shadow"]
			var bp: Vector3 = base[id][0]
			var hh: float = base[id][4]
			var ph := float(i) * 1.7
			var off := Vector3.ZERO
			var mod: Color = base[id][2]
			if fly.has(str(map[id])):
				var bob := sin(t * 3.2 + ph) * 0.09
				off.y += bob
				sh.scale = base[id][3] * (1.0 - bob * 0.8)
				s.scale = Vector3.ONE
			else:
				var br := sin(t * 2.4 + ph) * 0.022 # breathe: stretch up from the feet
				s.scale = Vector3(1.0 - br * 0.5, 1.0 + br, 1.0)
				off += lean * (br * hh)
			for p in plan:
				var dt: float = t - float(p[0])
				if dt < 0.0 or dt > 0.6:
					continue
				if p[1] == id: # lunge: 0.12 s out, 0.28 s back
					var tg: Vector3 = base[p[2]][0]
					var dir := Vector3(tg.x - bp.x, 0, tg.z - bp.z).normalized()
					var k := dt / 0.12 if dt < 0.12 else maxf(0.0, 1.0 - (dt - 0.12) / 0.28)
					off += dir * 0.7 * k * k * (3.0 - 2.0 * k)
				if p[2] == id and dt > 0.1 and dt < 0.4: # hit: flash + shake + small knock back
					var hk := 1.0 - (dt - 0.1) / 0.3
					mod = mod.lerp(Color(2.2, 2.0, 1.9), hk * 0.8)
					off.x += sin(dt * 90.0) * 0.06 * hk
					off.x += (0.12 if map_side[id] == 1 else -0.12) * hk
			s.position = bp + off
			sh.position = base[id][1] + Vector3(off.x, 0, off.z)
			s.modulate = mod
		for k in int(EV.get("fstep", "1")):
			await process_frame
		await RenderingServer.frame_post_draw
		root.get_texture().get_image().save_png(out.path_join("A_f%03d.png" % f))


# s233 B1: rolling ground - same th() as tests/s233_terrain.gdshader (change both). A fine near mesh over the arena +
# the old far Ground gets the same shader, sunk a little so the near mesh wins where both exist.
func _s233_h(p: Vector2) -> float:
	return fposmod(sin(p.dot(Vector2(127.1, 311.7))) * 43758.5453, 1.0)
func _s233_vn(p: Vector2) -> float:
	var i := p.floor()
	var f := p - i
	f = f * f * (Vector2(3, 3) - 2.0 * f)
	return lerpf(lerpf(_s233_h(i), _s233_h(i + Vector2(1, 0)), f.x), lerpf(_s233_h(i + Vector2(0, 1)), _s233_h(i + Vector2(1, 1)), f.x), f.y)
func _s233_fbm(p: Vector2) -> float:
	return _s233_vn(p) * 0.55 + _s233_vn(p * 2.03 + Vector2(5.1, 5.1)) * 0.3 + _s233_vn(p * 4.1 + Vector2(9.7, 9.7)) * 0.15
func _s233_th(p: Vector2, EV: Dictionary) -> float:
	if EV.get("meadow", "0") == "1": return _s237_th(p)
	var r := p.length() / 11.7
	var shelf := -0.04 - 0.11 * smoothstep(1.15, 1.6, r)
	var mound := float(EV.get("tamp", "1.6")) * smoothstep(1.35, 2.8, r) * pow(_s233_fbm(p * 0.07), 1.3)
	var dr := Vector2(p.x * 0.85, p.y + 2.0).length()
	var fall := float(EV.get("tfall", "0"))
	if fall > 0.0:
		var h1 := float(EV.get("thill1", "0"))
		if h1 > 0.0:
			var c1 := (dr - 95.0) / 26.0
			var c2 := (dr - 190.0) / 42.0
			var r1 := h1 * exp(-c1 * c1) * (0.45 + 0.55 * _s233_fbm(p * 0.016 + Vector2(3.3, 1.7)))
			var r2 := float(EV.get("thill2", "0")) * exp(-c2 * c2) * (0.45 + 0.55 * _s233_fbm(p * 0.009 + Vector2(7.1, 2.9)))
			var c3 := (dr - 140.0) / 22.0
			var r3 := float(EV.get("thill3", "0")) * exp(-c3 * c3) * (0.4 + 0.6 * _s233_fbm(p * 0.013 + Vector2(5.2, 8.4)))
			var vr := float(EV.get("tvroll", "0")) * smoothstep(38.0, 75.0, dr) * _s233_fbm(p * 0.022 + Vector2(1.9, 6.1))
			return shelf + mound - fall * smoothstep(float(EV.get("tf0", "16")), float(EV.get("tf1", "70")), dr) + r1 + r2 + r3 + vr - 30.0 * smoothstep(250.0, 330.0, dr)
		return shelf + mound - fall * smoothstep(float(EV.get("tf0", "16")), float(EV.get("tf1", "70")), dr) - 40.0 * smoothstep(90.0, 140.0, dr)
	return shelf + mound - 60.0 * smoothstep(20.0, 45.0, dr)

# s233 R11 (owner G2): lower camera like the owner painting. Sprites re-lean to the new pitch, feet stay put.

# s233 R13 (owner W1): one flat sea sheet in the valley, like FF's water001 (a flat mesh below the field) [DATA];
# the look (sky tint by angle, slow ripples, light shoreline) is ours [APPROX]. The shore needs the ground height, so the
# water shader reuses the terrain shader's own uniforms + th() (taken from its code: one formula, no copy to drift).
const WATER233 := """
uniform float wl = -22.0;
uniform vec3 deep = vec3(0.16, 0.36, 0.46);
uniform vec3 skyc = vec3(0.66, 0.80, 0.90);
uniform vec3 shore_c = vec3(0.78, 0.86, 0.80);
uniform float shore_w = 2.2;
uniform float glit = 0.0; // s234: glints like the owner's water ref (bright flecks, stretched across the view)
uniform float gsx = 0.7;
uniform float gsz = 0.18;
uniform float gthr = 0.55;
varying vec3 wp2;
void vertex() { wp2 = (MODEL_MATRIX * vec4(VERTEX, 1.0)).xyz; }
void fragment() {
	float g = th(wp2.xz);
	float depth = wl - g;
	if (depth < -0.05) { discard; }
	float fres = pow(1.0 - clamp(dot(NORMAL, VIEW), 0.0, 1.0), 2.2);
	vec2 q = wp2.xz * 0.08;
	float rip = vn(q + vec2(TIME * 0.05, TIME * 0.03)) * 0.6 + vn(q * 3.1 - vec2(TIME * 0.08, 0.0)) * 0.4;
	vec3 c = mix(deep, skyc, clamp(fres + (rip - 0.5) * 0.25, 0.0, 1.0));
	c += vec3(0.9, 0.85, 0.7) * smoothstep(0.82, 0.95, rip) * 0.12; // sparkles
	if (glit > 0.0) {
		vec2 g2 = vec2(wp2.x * gsx, wp2.z * gsz);
		float s1 = vn(g2 + vec2(TIME * 0.4, TIME * 0.15));
		float s2 = vn(g2 * 2.3 + vec2(-TIME * 0.3, 7.3));
		float gl = smoothstep(gthr, gthr + 0.12, s1 * s2 * 1.8) * (0.4 + 0.6 * fres);
		c += vec3(1.0, 0.98, 0.92) * gl * glit;
	}
	float sh = 1.0 - smoothstep(0.0, shore_w, depth);
	c = mix(c, shore_c, sh * 0.75);
	ALBEDO = c;
}
"""
func _s233_water(st: Node, EV: Dictionary) -> void:
	var old := st.find_child("Ground", true, false) as MeshInstance3D
	var tcode := (load("res://tests/s233_terrain.gdshader") as Shader).code
	var cut := tcode.find("void vertex()")
	var code := tcode.substr(0, cut).replace("render_mode specular_disabled;", "render_mode unshaded, cull_disabled;") + WATER233
	var sh := Shader.new()
	sh.code = code
	var m := _s233_mat(sh, old.material_override as ShaderMaterial, 0.0, 0.0, EV)
	var wl := float(EV["wl"])
	m.set_shader_parameter("wl", wl)
	for k in ["glit", "gsx", "gsz", "gthr"]:
		if EV.has(k):
			m.set_shader_parameter(k, float(EV[k]))
	for k in ["deep", "skyc", "shore_c"]:
		if EV.has("w" + k):
			var cc := Color(EV["w" + k])
			m.set_shader_parameter(k, Vector3(cc.r, cc.g, cc.b))
	var w := MeshInstance3D.new()
	w.name = "S233Sea"
	var pm := PlaneMesh.new()
	pm.size = Vector2(800, 520)
	w.mesh = pm
	w.material_override = m
	w.cast_shadow = GeometryInstance3D.SHADOW_CASTING_SETTING_OFF
	st.add_child(w)
	w.global_position = Vector3(0, wl, -280)
	print("s233 sea at ", wl)

# s233 R15 A2 (owner): a waterfall into the sea. Our own look [APPROX]: an owner rock as the cliff, a scrolling
# streak sheet, foam on the water and a mist card. Placed where the far ridge leaves the sea at x = wfall.
const FALL_NOISE := """
float h(vec2 p) { return fract(sin(dot(p, vec2(127.1, 311.7))) * 43758.5453); }
float vn(vec2 p) {
	vec2 i = floor(p); vec2 f = fract(p); f = f * f * (3.0 - 2.0 * f);
	return mix(mix(h(i), h(i + vec2(1.0, 0.0)), f.x), mix(h(i + vec2(0.0, 1.0)), h(i + vec2(1.0, 1.0)), f.x), f.y);
}
"""
const FALL_SHEET := """shader_type spatial;
render_mode unshaded, cull_disabled, depth_draw_never;
uniform vec3 c_deep = vec3(0.42, 0.62, 0.70);
uniform vec3 c_foam = vec3(0.93, 0.97, 1.0);
%s
void fragment() {
	vec2 uv = UV;
	float wob = (vn(vec2(uv.y * 4.0, TIME * 0.5)) - 0.5) * 0.12;
	float edge = smoothstep(0.0, 0.2, uv.x + wob) * smoothstep(1.0, 0.8, uv.x - wob);
	float s = vn(vec2(uv.x * 22.0, uv.y * 2.5 - TIME * 1.6)) * 0.6 + vn(vec2(uv.x * 55.0, uv.y * 6.0 - TIME * 2.4)) * 0.4;
	vec3 c = mix(c_deep, c_foam, smoothstep(0.3, 0.75, s));
	c = mix(c, c_foam, smoothstep(0.65, 1.0, uv.y));
	ALBEDO = c;
	ALPHA = edge * smoothstep(0.0, 0.05, uv.y) * (0.7 + 0.3 * s);
}
"""
const FALL_FOAM := """shader_type spatial;
render_mode unshaded, cull_disabled, depth_draw_never;
uniform float k = 0.85;
%s
void fragment() {
	float d = length((UV - 0.5) * 2.0);
	float n = vn(UV * 9.0 + vec2(TIME * 0.4, -TIME * 0.3)) * 0.6 + vn(UV * 23.0 - TIME * 0.5) * 0.4;
	ALBEDO = vec3(0.95, 0.98, 1.0);
	ALPHA = smoothstep(1.0, 0.2, d + (n - 0.5) * 0.5) * k * (0.55 + 0.45 * n);
}
"""

	mi.material_override = m
	mi.cast_shadow = GeometryInstance3D.SHADOW_CASTING_SETTING_OFF
	st.add_child(mi)
	mi.global_position = pos
	return mi

# s234 F1 (owner: grass looks like it floats on the slabs): top-down mask of every grass patch near the plaza -> the
# plaza shader paints soil there (slabs gone) with a dirty broken rim, so the grass grows out of earth.
func _s234_gmask(g: MeshInstance3D, c: Vector2, r: float, centres: Array, EV: Dictionary) -> void:
	var ext := float(EV.get("gmext", "16"))
	var res := 512
	var img := Image.create(res, res, false, Image.FORMAT_L8)
	var k := float(EV.get("gmk", "1.0")) # mask ellipse size vs the patch
	var n := 0
	for cc in centres:
		var cen: Vector2 = cc[0]
		if cen.distance_to(c) > r * 1.3: continue
		if (cc as Array).size() > 6 and cc[6] > 0.5 and EV.get("pgcrack", "0") == "1": continue
		var rx: float = cc[1] * k
		var rz: float = cc[2] * k
		var yaw: float = cc[3]
		var ca := cos(yaw); var sa := sin(yaw)
		var big := maxf(rx, rz)
		var x0 := int((cen.x - big + ext) / (2.0 * ext) * res); var x1 := int((cen.x + big + ext) / (2.0 * ext) * res)
		var y0 := int((cen.y - big + ext) / (2.0 * ext) * res); var y1 := int((cen.y + big + ext) / (2.0 * ext) * res)
		for py in range(maxi(y0, 0), mini(y1 + 1, res)):
			for px in range(maxi(x0, 0), mini(x1 + 1, res)):
				var w := Vector2((px + 0.5) / res * 2.0 * ext - ext, (py + 0.5) / res * 2.0 * ext - ext) - cen
				var lx := w.x * ca + w.y * sa
				var lz := -w.x * sa + w.y * ca
				var e := sqrt((lx / rx) * (lx / rx) + (lz / rz) * (lz / rz))
				var v := 1.0 - smoothstep(0.55, 1.0, e)
				if v > img.get_pixel(px, py).r: img.set_pixel(px, py, Color(v, v, v))
		n += 1
	var m := g.material_override as ShaderMaterial
	m.set_shader_parameter("use_gmask", true)
	m.set_shader_parameter("grass_mask", ImageTexture.create_from_image(img))
	m.set_shader_parameter("gm_ext", ext)
	m.set_shader_parameter("gm_cut", float(EV.get("gmcut", "0.45")))
	m.set_shader_parameter("gm_shadow", float(EV.get("gmsh", "0")))
	var sc := Color(EV.get("gmsoil", "#33261a"))
	m.set_shader_parameter("soil", Vector3(sc.r, sc.g, sc.b))
	print("s233 gmask patches ", n)


# s235 (owner AI picture 2): deeper blue sky + warm cloud tops, tinted apart. Blue-ness of each texel picks the tint.
const SKY235 := """
shader_type spatial;
render_mode unshaded, cull_disabled, fog_disabled;
uniform sampler2D tex : source_color, filter_linear_mipmap;
uniform vec3 blue_k = vec3(0.8, 0.9, 1.1);
uniform vec3 cloud_k = vec3(1.15, 1.0, 0.85);
uniform float bsharp = 4.0;
void fragment() {
	vec4 c = texture(tex, UV);
	if (c.a < 0.5) discard;
	float b = clamp((c.b - max(c.r, c.g * 0.9)) * bsharp, 0.0, 1.0);
	ALBEDO = mix(c.rgb * cloud_k, c.rgb * blue_k, b);

		m.set_shader_parameter("bsharp", float(EV.get("skys", "4.0")))

# s235: fake long sun shadow per unit = the sprite's silhouette lying flat on the ground from its feet
# (the plaza shading is baked, so real shadow maps do not show on it). ulong = alpha, ulen = length factor, udir = x,z.
func _s235_longshadow(st: Node, sprs: Array, EV: Dictionary) -> void:
	var dv: Array = str(EV.get("udir", "-1,0.35")).split_floats(",")
	var dir := Vector3(dv[0], 0.0, dv[1]).normalized()

		var sc := s.global_transform.basis.get_scale()
		var yv := dir * L * sc.y

		var xv := yv.cross(zv).normalized() * sc.x
		var p := s.global_position
		var bottom := p.y - (h * 0.5 - s.offset.y) * s.pixel_size * sc.y if s.centered else p.y
		sh.global_transform = Transform3D(Basis(xv, yv, zv), Vector3(p.x, 0.03, p.z))

# s235: owner Gemini tree sheets (scratch tests/s225/tree[ABC][0-2].png, keyed by tools/key_trees.py) as upright cards.
# Near trees frame the plaza left/right like owner AI picture 2; small hazy ones dot the far ridge.
# s238 round 4 (owner: "trees and bushes work like the grass"): the card sways in the wind like the grass patches
# (shader TIME, base fixed, top moves most, phase from world position). Billboard Y like the old Sprite3D.
const SWAY238 := """
shader_type spatial;
render_mode unshaded, cull_disabled, shadows_disabled, depth_prepass_alpha;
uniform sampler2D tex : source_color, filter_linear_mipmap;
uniform vec3 tint = vec3(1.0);
uniform float flip = 0.0;
uniform float h = 10.0;
uniform float amp = 0.02;
uniform float spd = 1.2;
uniform float bias = -1.0;
uniform float base = 0.4; // s238 r6 (owner): the trunk stays still, only the part above this height fraction sways // s238 r5: sharper mip pick (owner: bushes blurry next to the grass)
void vertex() {
	vec3 o = MODEL_MATRIX[3].xyz;
	float t = clamp((VERTEX.y / h - base) / (1.0 - base), 0.0, 1.0);
	float ph = o.x * 0.31 + o.z * 0.17;
	VERTEX.x += (sin(TIME * spd + ph) + 0.35 * sin(TIME * spd * 2.3 + ph * 1.7)) * amp * h * t * t;
	MODELVIEW_MATRIX = VIEW_MATRIX * mat4(vec4(normalize(cross(vec3(0.0, 1.0, 0.0), INV_VIEW_MATRIX[2].xyz)), 0.0), vec4(0.0, 1.0, 0.0, 0.0), vec4(normalize(cross(INV_VIEW_MATRIX[0].xyz, vec3(0.0, 1.0, 0.0))), 0.0), MODEL_MATRIX[3]);
	MODELVIEW_NORMAL_MATRIX = mat3(MODELVIEW_MATRIX);
}
void fragment() {
	vec4 c = texture(tex, vec2(mix(UV.x, 1.0 - UV.x, flip), UV.y), bias);
	ALBEDO = c.rgb * tint;
	ALPHA = c.a;
	ALPHA_SCISSOR_THRESHOLD = 0.5;
}
"""
var _s238_sway := 0.0 # s238: wind amount for tree/bush cards (tsway); 0 = old still Sprite3D
var _s238_sh: Shader = null
var _s238_shp := 0.0 # s238 r5: unsharp amount for tree/bush card pictures (tshp)
var _s238_bias := -1.0
var _s238_base := 0.45 # trunk share of a tree card that never moves (tbase)
var _s238_shp2 := 0.0 # fine-detail unsharp (tshp2)
# unsharp mask on the colour (alpha untouched): c + k * (c - blur(c))
func _s238_sharpen(im: Image, k: float, r: int) -> Image:
	im.convert(Image.FORMAT_RGBA8)
	var w := im.get_width(); var h := im.get_height()
	var b := im.duplicate() as Image
	b.resize(maxi(1, w / r), maxi(1, h / r), Image.INTERPOLATE_BILINEAR)
	b.resize(w, h, Image.INTERPOLATE_BILINEAR)
	var d := im.get_data(); var bd := b.get_data()
	for i in range(0, d.size(), 4):
		if d[i + 3] == 0: continue
		for c in 3:
			d[i + c] = clampi(int(d[i + c] + k * (d[i + c] - bd[i + c])), 0, 255)
	return Image.create_from_data(w, h, false, Image.FORMAT_RGBA8, d)
var _s238_tex := {}
func _s238_card(st: Node, file: String, at: Vector3, h: float, tint: Color, flip: bool) -> MeshInstance3D:
	if not _s238_tex.has(file):
		var im := Image.load_from_file(ProjectSettings.globalize_path(file))
		if im == null: return null
		if _s238_shp > 0.0: im = _s238_sharpen(im, _s238_shp, 3)
		if _s238_shp2 > 0.0: im = _s238_sharpen(im, _s238_shp2, 2)
		im.generate_mipmaps()
		_s238_tex[file] = ImageTexture.create_from_image(im)
	var t: ImageTexture = _s238_tex[file]
	if _s238_sh == null:
		_s238_sh = Shader.new(); _s238_sh.code = SWAY238
	var w := h * float(t.get_width()) / float(t.get_height())
	var q := MeshInstance3D.new()
	var qm := QuadMesh.new()
	qm.size = Vector2(w, h)
	qm.center_offset = Vector3(0, h * 0.5, 0)
	q.mesh = qm
	var m := ShaderMaterial.new()
	m.shader = _s238_sh
	m.set_shader_parameter("tex", t)
	m.set_shader_parameter("tint", Vector3(tint.r, tint.g, tint.b))
	m.set_shader_parameter("flip", 1.0 if flip else 0.0)
	m.set_shader_parameter("h", h)
	m.set_shader_parameter("bias", _s238_bias)
	var small := h < 4.0 # bushes sway a bit more than trees, like the grass
	m.set_shader_parameter("amp", _s238_sway * (1.8 if small else 1.0))
	m.set_shader_parameter("spd", 1.6 if small else 1.1)
	m.set_shader_parameter("base", 0.15 if small else _s238_base)
	q.material_override = m
	q.cast_shadow = GeometryInstance3D.SHADOW_CASTING_SETTING_OFF
	st.add_child(q)
	q.global_position = at - Vector3(0, h * 0.03, 0)
	return q

func _s236_dump(n: Node, d: int, maxd: int) -> void:
	var vis := (n as Node3D).visible if n is Node3D else true
	var extra := ""
	if n is MeshInstance3D and (n as MeshInstance3D).mesh: extra = (n as MeshInstance3D).mesh.get_class()
	if n is MultiMeshInstance3D and (n as MultiMeshInstance3D).multimesh: extra = "MM %d" % (n as MultiMeshInstance3D).multimesh.instance_count
	print("  ".repeat(d), n.name, " [", n.get_class(), "] vis=", vis, " kids=", n.get_child_count(), " ", extra)
	if d < maxd:
		var seen := {}
		for c in n.get_children():
			var key: String = c.get_class() + ":" + String(c.name).rstrip("0123456789")
			seen[key] = int(seen.get(key, 0)) + 1
			if int(seen[key]) <= 3: _s236_dump(c, d + 1, maxd)
		for key in seen.keys():
			if int(seen[key]) > 3: print("  ".repeat(d + 1), "… ", key, " x", seen[key])

var _s236_tex := {} # image hash → res path (pass 2) / saved (pass 1)

func _s236_keep(c: Node) -> bool:
	if c is Node3D and not (c as Node3D).visible: return false
	if c is Camera3D or c is WorldEnvironment or c is DirectionalLight3D: return false
	var nm := String(c.name)
	if nm.begins_with("Tile") or nm.begins_with("Rim") or nm.begins_with("Ub_") or (nm.begins_with("U") and nm.substr(1).is_valid_int()): return false
	if c.has_meta("unit_fx") or c.has_meta("follow") or shadows.has(c): return false
	return true

func _s236_prune(n: Node) -> void:
	for c in n.get_children():
		if c is Node3D and not (c as Node3D).visible:
			n.remove_child(c); c.queue_free()
		else:
			_s236_prune(c)

func _s236_own(n: Node, owner_node: Node) -> void:
	for c in n.get_children():
		c.owner = owner_node
		if c.scene_file_path != "": c.scene_file_path = "" # make local: the sample changed nodes inside these instances
		if c.get_script() != null: print("S236 script ", c.name, " ", (c.get_script() as Script).resource_path)
		_s236_own(c, owner_node)

func _s236_res(r: Resource, dir: String, pass_n: int) -> Resource:
	if r == null: return r
	if _s236_seen.has(r): return _s236_seen[r]
	var path := r.resource_path
	var builtin := path == "" or path.contains("::")
	var out: Resource = r
	if r is Texture2D and not (r is ViewportTexture):
		var img := (r as Texture2D).get_image()
		var nm := ("t_" + str(hash(img.get_data())) + "_%dx%d" % [img.get_width(), img.get_height()]) if builtin else path.get_file().get_basename()
		var f := dir + "/tex/" + nm + ".webp"
		if pass_n == 1:
			if not _s236_tex.has(nm):
				if img.is_compressed(): img.decompress()
				if img.has_mipmaps(): img.clear_mipmaps()
				var lossy := img.get_width() * img.get_height() > 300000 and img.get_format() != Image.FORMAT_L8 and img.get_format() != Image.FORMAT_R8
				img.save_webp(ProjectSettings.globalize_path(f), lossy, 0.9); _s236_tex[nm] = f
		else:
			var t: Variant = load(f)
			if t == null: print("S236 MISSING ", f)
			else: out = t
		_s236_seen[r] = out
		return out
	_s236_seen[r] = out
	if not builtin:
		if pass_n == 2 and r is Shader and path.begins_with("res://tests/"): # sample shader → the arena folder
			var f := dir + "/shader/" + path.get_file()
			var fa := FileAccess.open(ProjectSettings.globalize_path(f), FileAccess.WRITE)

# ---- s237 A1 (owner AI overhead picture): real meadow layout. _s237_th mirrors th237() in tests/s233_terrain.gdshader - change both.
func _s237_th(p: Vector2) -> float:
	var r := p.length() / 11.7
	var shelf := -0.04 - 0.11 * smoothstep(1.15, 1.6, r)
	var mnd := 1.0 * smoothstep(1.35, 2.8, r) * pow(_s233_fbm(p * 0.07), 1.3)
	var zf := -105.0 + _s238_lk - 15.0 * _s233_fbm(Vector2(p.x * 0.03, 3.1))
	var plain := -0.3 * smoothstep(-14.0, -60.0, p.y) - 2.4 * smoothstep(-60.0, zf, p.y)
	var ex := (p.x + 15.0) / 150.0
	var ez := (p.y + 172.0 - _s238_lk) / 72.0
	var e := ex * ex + ez * ez
	var basin := -6.5 * (1.0 - smoothstep(0.5, 1.0, e))
	var hr := 13.0 * smoothstep(18.0, 85.0, p.x) * smoothstep(-15.0, -45.0, p.y) * (0.55 + 0.45 * _s233_fbm(p * 0.02 + Vector2(4.2, 1.3))) * smoothstep(0.9, 1.4, e) * (1.0 - _s238_hrc * smoothstep(-185.0, -215.0, p.y))
	var a := Vector2(-16.0, -46.0)
	var ab := Vector2(-62.0, -122.0) - a
	var ap := p - a
	var t := clampf(ap.dot(ab) / ab.dot(ab), 0.0, 1.0)
	var d := (ap - ab * t).length()
	var sd := ab.x * ap.y - ab.y * ap.x
	var w := 11.0 if (sd > 0.0 or t < 0.05) else 26.0
	var ridge := _s238_rdg * (4.5 + 2.5 * _s233_fbm(p * 0.05 + Vector2(8.8, 2.2))) * (1.0 - smoothstep(2.5, w, d)) * (1.0 - 0.6 * smoothstep(0.75, 1.0, t)) * smoothstep(0.0, 0.06, t + 0.02)
	var dm := Vector2((p.x + 15.0) * 0.8, p.y + 172.0 - _s238_lk).length()
	var back := smoothstep(-40.0, -110.0, p.y) * _s237_mtk
	var mt := (60.0 * smoothstep(150.0, 270.0, dm) * (0.35 + 0.65 * _s233_fbm(p * 0.011 + Vector2(2.7, 9.1))) + 25.0 * smoothstep(110.0, 200.0, dm) * _s233_fbm(p * 0.025 + Vector2(6.3, 1.1))) * back
	return shelf + mnd + plain + basin + hr + ridge + mt

# distance to the nearest dirt path edge (negative = on it); same paths as path237() in the shader (no noise here)
var _s238_rdg := 1.0 # s238: ridge scale (rdg) - twin of the shader uniform rdg
var _s238_hrc := 0.0 # s238: right hills cut behind the lake (hrc) - twin of the shader uniform hrc
var _s238_lk := 0.0 # s238: lake shift in z (lkz) - twin of the shader uniform lkz
var _s237_mtk := 1.0 # s237: terrain mountain scale (mtk); low when the silhouette layers stand behind
var _s237_np := false # s237: owner removed the dirt paths (tpath=0) -> nothing avoids them
func _s237_path(p: Vector2) -> float:
	if _s237_np: return 99.0
	var pz := -1.0 + 1.8 * sin(p.x * 0.05 + 0.6)
	var d1 := absf(p.y - pz) - 0.8 + (99.0 if absf(p.x) >= 95.0 else 0.0)
	var px := 3.0 + 14.0 * smoothstep(-14.0, -100.0, p.y) + 2.5 * sin(p.y * 0.13)
	var d2 := absf(p.x - px) - 0.7 + (0.0 if (p.y <= -11.0 and p.y >= -104.0) else 99.0)
	return minf(d1, d2)

# s237 (owner: layered mountains behind the lake like the AI picture): N silhouette quads, near -> far, each lighter/bluer,
# feet below the water line so the mist hides them. Own procedural shader (no outside asset).
const LAYER237 := """
shader_type spatial;
render_mode unshaded, cull_disabled, shadows_disabled, fog_disabled;
uniform vec3 top_c = vec3(0.4, 0.55, 0.42);
uniform vec3 foot_c = vec3(0.7, 0.78, 0.76);
uniform float base = 0.35;
uniform float amp = 0.45;
uniform float freq = 3.0;
uniform float peak = 0.0;
uniform float seed = 0.0;
float h(float x) { return fract(sin(x * 127.1 + seed) * 43758.5453); }
float n1(float x) { float i = floor(x); float f = fract(x); f = f * f * (3.0 - 2.0 * f); return mix(h(i), h(i + 1.0), f); }
float ridge(float u) {
	float x = u * freq;
	float r = n1(x) * 0.55 + n1(x * 2.1 + 3.0) * 0.28 + n1(x * 4.7 + 7.0) * 0.12 + n1(x * 11.0) * 0.05;
	r = mix(r, 1.0 - abs(r * 2.0 - 1.0), peak); // sharper peaks for the far range
	return base + amp * r;
}
void fragment() {
	float y = 1.0 - UV.y;
	float rr = ridge(UV.x);
	if (y > rr) { discard; }
	float e = 0.004;
	float sl = (ridge(UV.x + e) - ridge(UV.x - e)) / (2.0 * e);
	float lit = clamp(0.5 + sl * 0.08, 0.0, 1.0); // sun from the right: right-facing slopes lighter
	float t = clamp(y / max(rr, 0.001), 0.0, 1.0);
	vec3 c = mix(foot_c, top_c, smoothstep(0.0, 0.85, t));
	c *= mix(0.86, 1.1, lit);
	c *= 0.94 + 0.12 * n1(UV.x * 90.0 + y * 40.0); // faint forest/rock grain
	ALBEDO = c;
}
"""

var _s238_trk := {} # trunk file -> [anchors (Vector2 u,v from top), tips, trunk axis u, aspect w/h]
func _s238_leaf_atlas() -> ImageTexture:
	if _s238_leaf_tex != null: return _s238_leaf_tex
	var at := Image.create(LEAF_CW * 4, LEAF_CH * 3, false, Image.FORMAT_RGBA8)
	for i in 12:
		var im := Image.load_from_file(ProjectSettings.globalize_path("res://tests/s238/sprig_A%d_key_2026-11-15.png" % i))
		im.convert(Image.FORMAT_RGBA8)
		var x := (i % 4) * LEAF_CW + (LEAF_CW - im.get_width()) / 2
		var y := (i / 4) * LEAF_CH + LEAF_CH - 12 - im.get_height()
		at.blit_rect(im, Rect2i(0, 0, im.get_width(), im.get_height()), Vector2i(x, y))
	if _s238_shp > 0.0: at = _s238_sharpen(at, _s238_shp, 3)
	if _s238_shp2 > 0.0: at = _s238_sharpen(at, _s238_shp2, 2)
	at.generate_mipmaps()
	_s238_leaf_tex = ImageTexture.create_from_image(at)
	return _s238_leaf_tex

func _s238_trunk_info(file: String) -> Array:
	if _s238_trk.has(file): return _s238_trk[file]
	var im := Image.load_from_file(ProjectSettings.globalize_path(file))
	var w := im.get_width(); var h := im.get_height()
	var ax := 0.0; var na := 0
	for y in range(int(h * 0.85), h, 4):
		for x in range(0, w, 4):
			if im.get_pixel(x, y).a > 0.5: ax += x; na += 1
	ax = ax / maxf(1.0, na)
	var pts: Array[Vector2] = []
	var tips: Array[Vector2] = []
	for y in range(0, int(h * 0.7), 6):
		for x in range(0, w, 6):
			if im.get_pixel(x, y).a > 0.5: pts.append(Vector2(float(x) / w, float(y) / h))
	for cx in 8: # tips: top-most opaque point per 1/8 width column
		var best := Vector2(-1, 2)
		for q in pts:
			if int(q.x * 8.0) == cx and q.y < best.y: best = q
		if best.x >= 0.0: tips.append(best)
	var r := [pts, tips, ax / w, float(w) / float(h)]
	_s238_trk[file] = r
	return r

func _s238_mm(st: Node, at: Vector3, items: Array, tint: Color, amp: float, spd: float, nm: String) -> void:
	if _s238_leaf_sh == null:
		_s238_leaf_sh = Shader.new(); _s238_leaf_sh.code = LEAF238
	var mm := MultiMesh.new()
	mm.transform_format = MultiMesh.TRANSFORM_3D
	mm.use_custom_data = true
	var qm := QuadMesh.new()
	qm.size = Vector2(1, 1)
	qm.center_offset = Vector3(0, 0.5, 0)
	mm.mesh = qm
	mm.instance_count = items.size()
	for i in items.size():
		var it: Array = items[i]
		var sz: Vector2 = it[1]
		mm.set_instance_transform(i, Transform3D(Basis.from_scale(Vector3(sz.x, sz.y, 1.0)), it[0]))
		mm.set_instance_custom_data(i, Color(it[2], it[3], it[4], it[5]))
	var mi := MultiMeshInstance3D.new()
	mi.name = nm
	mi.multimesh = mm
	var m := ShaderMaterial.new()
	m.shader = _s238_leaf_sh
	m.set_shader_parameter("tex", _s238_leaf_atlas())
	m.set_shader_parameter("tint", Vector3(tint.r, tint.g, tint.b))
	m.set_shader_parameter("amp", amp)
	m.set_shader_parameter("spd", spd)
	m.set_shader_parameter("bias", _s238_bias)
	mi.material_override = m
	mi.cast_shadow = GeometryInstance3D.SHADOW_CASTING_SETTING_OFF
	st.add_child(mi)
	mi.global_position = at

var _s238_pt_n := 0
func _s238_ptree(st: Node, EV: Dictionary, at: Vector3, h: float, rng: RandomNumberGenerator) -> void:
	var trunks := ["B0", "B1", "B2", "C0", "C1", "C2"]
	var tf := "res://tests/s238/trunk_%s_key_2026-11-15.png" % trunks[rng.randi() % trunks.size()]
	var flip := rng.randf() < 0.5
	var info := _s238_trunk_info(tf)
	var tt: Array = str(EV.get("trkt", "0.8,0.78,0.74")).split_floats(",")
	var tq := _s238_card(st, tf, at, h, Color(tt[0], tt[1], tt[2]), flip)
	if tq == null: return
	var tm := (tq.material_override as ShaderMaterial).duplicate() as ShaderMaterial
	tm.set_shader_parameter("amp", 0.0) # trunk + branches stay still
	tq.material_override = tm
	tq.name = "Trunk%d" % _s238_pt_n
	var w: float = h * float(info[3])
	var pts: Array = info[0]
	var tips: Array = info[1]
	var ax: float = info[2]
	var lo: Array = str(EV.get("leafn", "32,48")).split_floats(",")
	var n := rng.randi_range(int(lo[0]), int(lo[1]))
	var ls := float(EV.get("leafs", "0.30"))
	var items := []
	var cand: Array[Vector2] = []
	for q in tips: cand.append(q)
	var guard := 0
	while cand.size() < n and guard < n * 60 and pts.size() > 0:
		guard += 1
		var q: Vector2 = pts[rng.randi() % pts.size()]
		var wgt := 0.25 + 1.6 * absf(q.x - ax) + 0.9 * (1.0 - q.y / 0.7)
		if rng.randf() * 1.6 < wgt: cand.append(q)
	for q in cand:
		var u := (1.0 - q.x) if flip else q.x
		var cx := (u - 0.5) * w
		var cy := -h * 0.03 + (1.0 - q.y) * h
		var cw := h * ls * rng.randf_range(0.7, 1.15)
		var ch := cw * float(LEAF_CH) / float(LEAF_CW)
		var dz := rng.randf_range(-0.9, 0.9)
		var shade := lerpf(0.72, 1.05, (dz + 0.9) / 1.8) * lerpf(0.9, 1.08, clampf(1.0 - q.y / 0.62, 0.0, 1.0))
		var off := Vector3(cx + rng.randf_range(-0.04, 0.04) * h, cy - ch * 0.5 + rng.randf_range(-0.03, 0.03) * h, dz)
		items.append([off, Vector2(cw, ch), float(rng.randi() % 12), 1.0 if rng.randf() < 0.5 else 0.0, shade, rng.randf_range(0.6, 1.0)])
	var lt: Array = str(EV.get("leaft", "0.92,0.98,0.9")).split_floats(",")
	_s238_mm(st, at, items, Color(lt[0], lt[1], lt[2]), _s238_sway, 1.2, "Leaves%d" % _s238_pt_n)
	_s238_pt_n += 1

func _s238_pbush(st: Node, at: Vector3, bh: float, tint: Color, rng: RandomNumberGenerator) -> void:
	var k := rng.randi_range(3, 5)
	var items := []
	for i in k:
		var cw := bh * rng.randf_range(1.0, 1.35) * (1.15 if i == 0 else 1.0)
		var ch := cw * float(LEAF_CH) / float(LEAF_CW)
		var a := rng.randf() * TAU
		var r := 0.0 if i == 0 else bh * rng.randf_range(0.25, 0.55)
		var off := Vector3(cos(a) * r, -ch * 0.04, sin(a) * r * 0.6)
		var shade := lerpf(0.78, 1.05, clampf(0.5 + off.z / (bh * 0.66), 0.0, 1.0))
		items.append([off, Vector2(cw, ch), float(rng.randi() % 12), 1.0 if rng.randf() < 0.5 else 0.0, shade, 1.0])
	_s238_mm(st, at, items, tint, _s238_sway * 1.8, 1.6, "Bush%d" % _s238_pt_n)
	_s238_pt_n += 1

func _s238_in(p: Vector2, poly: Array) -> bool:
	var c := false
	var j := poly.size() - 1
	for i in poly.size():
		var a: Vector2 = poly[i]
		var b: Vector2 = poly[j]
		if ((a.y > p.y) != (b.y > p.y)) and (p.x < (b.x - a.x) * (p.y - a.y) / (b.y - a.y) + a.x): c = not c
		j = i
	return c

# ---- s238 D (owner: "the picture already has them, use it"): S1 near-right hill + spire, S2 left range, S3 far range,
# cut from the owner's AI picture (art/s238/s1..s3.png, 300 px mirrored side pads, picture rows 0-249, editor UI painted out).
# Each card faces the GAME camera and is sized so the picture maps 1:1 onto the frame: picture x u0 -> screen centre,
# picture row 212 (lake far shore = owner red line) -> the water line, scale s game px per picture px. The water hides the feet.
const CARD238 := """
shader_type spatial;
render_mode unshaded, cull_disabled, shadows_disabled, fog_disabled, depth_draw_never;
uniform sampler2D tex : source_color, filter_linear_mipmap;
uniform vec3 tint = vec3(1.0);
uniform float con = 1.0;
void fragment() { vec4 c = texture(tex, UV); ALBEDO = clamp((c.rgb - 0.5) * con + 0.5, 0.0, 1.0) * tint; ALPHA = c.a; }
"""
func _s238_cards(st: Node, EV: Dictionary) -> void:
	var gc: Array = str(EV.get("gcam", "0,6.8,29.4,8.5,19.2")).split_floats(",")
	var cp := Vector3(gc[0], gc[1], gc[2])
	var pr := deg_to_rad(float(gc[3]))
	var B := Basis.from_euler(Vector3(-pr, 0, 0))
	var fwd := -B.z
	var thf := tan(deg_to_rad(float(gc[4])) * 0.5)
	var sc := float(EV.get("mls", "0.85"))
	var u0 := float(EV.get("mlu0", "1037"))
	var vs := float(EV.get("mlv0", "212"))
	var wl := float(EV.get("wl", "-5.2"))
	var tn: Array = str(EV.get("mltint", "1,1,1")).split_floats(",")
	var PAD := float(EV.get("mlpad", "300")) # side pad columns in the card image
	var pre := str(EV.get("mlset", "")) # s238 round 3: "fh_" = owner's earlier far_hills_v3 cut into layers
	# one shared shore row so the layers line up exactly as in the picture
	var rel := Vector3(0, wl, cp.z - 300.0 * cos(pr)) - cp
	var ys := 540.0 - rel.dot(B.y) / rel.dot(fwd) * (540.0 / thf) + float(EV.get("mldy", "0"))
	var sh := Shader.new()
	sh.code = CARD238
	var n := 0
	for c in [["s3", 316.0], ["s2", 308.0], ["s1", 300.0]]:
		var img := Image.load_from_file(ProjectSettings.globalize_path("res://art/s238/%s%s.png" % [pre, c[0]]))
		if img == null: continue
		img.generate_mipmaps()
		var D: float = c[1]
		var mpp := 2.0 * D * thf / 1080.0
		var x0 := 960.0 + (-PAD - u0) * sc
		var x1 := 960.0 + (float(img.get_width()) - PAD - u0) * sc
		var y0 := ys + (0.0 - vs) * sc
		var y1 := ys + (float(img.get_height()) - vs) * sc
		var q := MeshInstance3D.new()
		q.name = "S238Layer_" + str(c[0]).to_upper()
		var qm := QuadMesh.new()
		qm.size = Vector2((x1 - x0) * mpp, (y1 - y0) * mpp)
		q.mesh = qm
		var m := ShaderMaterial.new()
		m.shader = sh
		m.set_shader_parameter("tex", ImageTexture.create_from_image(img))
		m.set_shader_parameter("tint", Vector3(tn[0], tn[1], tn[2]))
		m.set_shader_parameter("con", float(EV.get("mlcon", "1.0")))
		q.material_override = m
		q.cast_shadow = GeometryInstance3D.SHADOW_CASTING_SETTING_OFF
		st.add_child(q)
		q.global_transform = Transform3D(B, cp + fwd * D + B.x * ((x0 + x1) * 0.5 - 960.0) * mpp - B.y * ((y0 + y1) * 0.5 - 540.0) * mpp)
		n += 1
		print("s238 card ", c[0], " D ", D, " size ", qm.size, " at ", q.global_position)
	print("s238 cards ", n, " shore row ", ys)
