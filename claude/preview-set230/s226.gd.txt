func _card(st: Node, file: String, at: Vector3, width: float, tint: Color) -> void:
	var t: Variant = load(file)
	if not (t is Texture2D):
		return
	var sp := Sprite3D.new()
	sp.texture = t
	sp.pixel_size = width / float((t as Texture2D).get_width())
	sp.billboard = BaseMaterial3D.BILLBOARD_FIXED_Y
	sp.alpha_cut = SpriteBase3D.ALPHA_CUT_DISCARD
	sp.shaded = false
	sp.modulate = tint
	sp.cast_shadow = GeometryInstance3D.SHADOW_CASTING_SETTING_OFF
	sp.position = at + Vector3(0, width * float((t as Texture2D).get_height()) / float((t as Texture2D).get_width()) * 0.5 - 0.12, 0)
	st.add_child(sp)

func _b6(st: Node, cam: Camera3D, EV: Dictionary) -> void:
	# 1 ground: shift the yellow-green toward the ref's blue-green (74,127,111) — a tint cannot add blue the texture lacks
	var teal := float(EV.get("teal", "0.5"))
	for mi in st.find_children("*", "MeshInstance3D", true, false):
		var mo: Material = (mi as MeshInstance3D).material_override
		if mo is ShaderMaterial and (mo as ShaderMaterial).get_shader_parameter("tint") != null:
			var sh := Shader.new()
			sh.code = (mo as ShaderMaterial).shader.code.replace("ALBEDO = c * tint;", "float l = dot(c, vec3(0.3, 0.59, 0.11));
	c = mix(c, l * vec3(0.78, 1.32, 1.18), %s);
	ALBEDO = c * tint;" % str(teal))
			(mo as ShaderMaterial).shader = sh
	# 2 plaza: warm beige like the ref's dirt path, a bit darker (owner tone script on the Plaza node)
	for n in st.find_children("Plaza", "", true, false):
		if "tone" in n:
			var pt: Array = str(EV.get("plaza", "0.66,0.6,0.48")).split_floats(",")
			n.tone = Color(pt[0], pt[1], pt[2])
	# 3 grass tufts: a ring at the plaza rim (back + sides), never inside the board
	var rng := RandomNumberGenerator.new()
	rng.seed = 22501
	var files := ["grass0", "grass1", "grass2", "grass3", "grass4", "grass5", "grass_big"]
	var gtint := Color(0.62, 0.74, 0.66)
	var ring_n := int(EV.get("tufts", "70"))
	for i in ring_n:
		var a := rng.randf_range(PI * 0.92, PI * 2.08) # back half + the two sides
		var r := rng.randf_range(15.0, 21.0)
		var at := Vector3(cos(a) * r * 1.12, 0, sin(a) * r - 1.5)
		var w := rng.randf_range(float(EV.get("tw0", "0.7")), float(EV.get("tw1", "1.4")))
		_card(st, "res://art/s217/%s.png" % files[rng.randi() % files.size()], at, w, gtint * rng.randf_range(0.85, 1.05))
	# 4 foreground framing: big tufts at the bottom corners, soft in the tilt-shift band (ref: grass between camera and field)
	if EV.get("fg", "1") == "1": # the view's bottom edge meets the ground at z ~7, half-width ~6 m there
		for fx in [[-7.4, 7.6, 2.2], [-5.6, 8.4, 1.6], [7.4, 7.8, 2.1], [5.8, 8.6, 1.5]]:
			_card(st, "res://art/s217/%s.png" % files[rng.randi() % files.size()], Vector3(fx[0], 0, fx[1]), fx[2], gtint * 0.72)

# s229: the plaza looked like it floats — no cast shadows in Compatibility, a clean cut edge, bright even grass.
# Fake contact shadow + darker grass band (one flat ring, unlit, alpha) + small tufts sitting ON the rim.
func _rim(st: Node, EV: Dictionary) -> void:
	var box := AABB()
	var first := true
	for n in st.find_children("Plaza", "", true, false):
		for mi in n.find_children("*", "MeshInstance3D", true, false):
			var b: AABB = (mi as MeshInstance3D).global_transform * (mi as MeshInstance3D).get_aabb()
			box = b if first else box.merge(b)
			first = false
	if first:
		return
	var c := box.get_center()
	var r := minf(box.size.x, box.size.z) * 0.5 * float(EV.get("rimr", "0.97"))
	var top := box.end.y
	var outer := r * float(EV.get("rimw", "1.45"))
	print("s229 plaza centre ", c, " r ", r, " top ", top, " bottom ", box.position.y)
	var mi := MeshInstance3D.new()
	var pm := PlaneMesh.new()
	pm.size = Vector2(outer * 2.0, outer * 2.0)
	mi.mesh = pm
	var sh := Shader.new()
	sh.code = """shader_type spatial;
render_mode unshaded, blend_mix, depth_draw_never, cull_disabled;
uniform float inner; uniform float a_edge; uniform float a_band; uniform vec3 col;
void fragment() {
	float d = length(UV - 0.5) * 2.0;
	float edge = 1.0 - smoothstep(inner, inner + 0.07, d);
	float band = 1.0 - smoothstep(inner, 1.0, d);
	float a = max(a_edge * edge, a_band * band) * step(inner * 0.985, d);
	ALBEDO = col; ALPHA = a;
}"""
	var m := ShaderMaterial.new()
	m.shader = sh
	m.set_shader_parameter("inner", r / outer)
	m.set_shader_parameter("a_edge", float(EV.get("rima", "0.6")))
	m.set_shader_parameter("a_band", float(EV.get("rimb", "0.3")))
	m.set_shader_parameter("col", Vector3(1, 0, 0) if EV.get("rimdbg", "0") == "1" else Vector3(0.05, 0.07, 0.04))
	mi.material_override = m
	mi.cast_shadow = GeometryInstance3D.SHADOW_CASTING_SETTING_OFF
	mi.position = Vector3(c.x, float(EV.get("rimy", "0.02")), c.z)
	st.add_child(mi)
	# small tufts on the rim, back half + sides (the front rim is under the foreground grass strip)
	var rng := RandomNumberGenerator.new()
	rng.seed = 22901
	var files := ["grass0", "grass1", "grass2", "grass3", "grass4", "grass5"]
	for i in int(EV.get("rimt", "90")):
		var a := rng.randf_range(PI * 0.9, PI * 2.1)
		var rr := r * rng.randf_range(0.95, 1.06)
		_card(st, "res://art/s217/%s.png" % files[rng.randi() % files.size()], Vector3(c.x + cos(a) * rr, 0.0, c.z + sin(a) * rr), rng.randf_range(0.35, 0.75), Color(0.5, 0.62, 0.52) * rng.randf_range(0.85, 1.05))

# s230 "baked" shade (FF Re bakes lightmaps; Compatibility has no live sun shadows): once per shot, rasterise every
# stone mesh into a top-down height grid, cast it along the sun direction (shadow) + blur its footprint (contact AO),
# then lay the result over plaza + grass as one multiply plane. Args: shd shadow strength · sao AO strength ·
# sres grid px · sext area half-size (units) · sblur shadow softness px · saor AO radius px · sy plane height.
func _s230_shade(st: Node, EV: Dictionary) -> void:
	if EV.get("slist", "0") == "1":
		for q in st.find_children("*", "MeshInstance3D", true, false):
			var q3 := q as Node3D
			if q3.is_visible_in_tree(): print("s230 mesh ", q3.get_path(), " at ", q3.global_position.round())
	if EV.get("skeep", "0") == "0": # hide the set 226 fake shadow cards (the bake replaces them)
		for q in st.find_children("Shadow*", "", true, false):
			(q as Node3D).visible = false
	var res := int(EV.get("sres", "512"))
	var ext := float(EV.get("sext", "26.0"))
	var hmap := PackedFloat32Array()
	hmap.resize(res * res)
	var n_v := 0
	for n in st.find_children("*", "", true, false):
		var nm := String(n.name)
		if not (nm.begins_with("Stone") or nm.begins_with("Fallen") or nm.begins_with("Rock")):
			continue
		for mi in n.find_children("*", "MeshInstance3D", true, false):
			var m := mi as MeshInstance3D
			if m.mesh == null or not m.is_visible_in_tree():
				continue
			var xf := m.global_transform
			for si in m.mesh.get_surface_count():
				var verts: PackedVector3Array = m.mesh.surface_get_arrays(si)[Mesh.ARRAY_VERTEX]
				var step := maxi(1, verts.size() / 20000)
				for vi in range(0, verts.size(), step):
					var p := xf * verts[vi]
					var gx := int((p.x + ext) / (2.0 * ext) * res)
					var gz := int((p.z + ext) / (2.0 * ext) * res)
					if gx < 0 or gz < 0 or gx >= res or gz >= res:
						continue
					var k := gz * res + gx
					if p.y > hmap[k]:
						hmap[k] = p.y
					n_v += 1
	# fill small holes between sampled vertices (3x3 max)
	var gnode := st.find_child("S230Ground", true, false) as MeshInstance3D
	var gtop := 0.0
	var gr := 0.0
	var gc := Vector2.ZERO
	if gnode != null:
		gtop = gnode.global_position.y + 0.15
		gr = (gnode.mesh as CylinderMesh).top_radius
		gc = Vector2(gnode.global_position.x, gnode.global_position.z)
	var h2 := hmap.duplicate()
	for z in range(1, res - 1):
		for x in range(1, res - 1):
			var mx := hmap[z * res + x]
			for dz in [-1, 0, 1]:
				for dx in [-1, 0, 1]:
					mx = maxf(mx, hmap[(z + dz) * res + x + dx])
			h2[z * res + x] = mx
	hmap = h2
	var sun: DirectionalLight3D = st.sun
	if EV.has("sunel"): # same rotation the look code applies later
		sun.rotation_degrees = Vector3(-float(EV["sunel"]), float(EV.get("sunaz", "-50")), 0.0)
	var ldir := -sun.global_transform.basis.z # direction the light travels
	var horiz := Vector2(ldir.x, ldir.z)
	var elev_tan := absf(ldir.y) / maxf(horiz.length(), 0.001)
	horiz = horiz.normalized()
	var shadow := PackedFloat32Array()
	shadow.resize(res * res)
	var occ := PackedFloat32Array()
	occ.resize(res * res)
	var cell := 2.0 * ext / res
	for z in res:
		for x in res:
			var wx := (float(x) + 0.5) / res * 2.0 * ext - ext
			var wz := (float(z) + 0.5) / res * 2.0 * ext - ext
			var base := gtop if gnode != null and Vector2(wx, wz).distance_to(gc) < gr else 0.0
			var h := hmap[z * res + x] - base # s231: height above the ground it stands on
			if h <= 0.05:
				continue
			occ[z * res + x] = 1.0
			var length_cells := (h / elev_tan) / cell
			var steps := int(minf(length_cells, res * 0.5))
			for s in steps:
				var sx := int(x + horiz.x * s)
				var sz := int(z + horiz.y * s)
				if sx < 0 or sz < 0 or sx >= res or sz >= res:
					break
				shadow[sz * res + sx] = 1.0
	var img := Image.create(res, res, false, Image.FORMAT_RGB8)
	var img_ao := Image.create(res, res, false, Image.FORMAT_L8)
	for z in res:
		for x in res:
			var v := shadow[z * res + x]
			img.set_pixel(x, z, Color(v, v, v))
			img_ao.set_pixel(x, z, Color(occ[z * res + x], 0, 0))
	img = _s230_blur(img, float(EV.get("sblur", "3")))
	img_ao = _s230_blur(img_ao, float(EV.get("saor", "10")))
	var sd := float(EV.get("shd", "0.45"))
	var ao := float(EV.get("sao", "0.55"))
	var out := Image.create(res, res, false, Image.FORMAT_RGB8)
	for z in res:
		for x in res:
			var occ_here := occ[z * res + x]
			var a := clampf(img_ao.get_pixel(x, z).r * 2.0, 0.0, 1.0) * (1.0 - occ_here)
			var f := (1.0 - sd * img.get_pixel(x, z).r) * (1.0 - ao * a)
			out.set_pixel(x, z, Color(f, f, f))
	out.generate_mipmaps()
	out.save_png(ProjectSettings.globalize_path("res://tests/s225/_s230_shade_dbg.png"))
	if EV.get("sdbg", "0") == "1": # s231 find-cause: footprint red, shadow blue
		for z in res:
			for x in res:
				var o := occ[z * res + x]
				var sv := shadow[z * res + x]
				out.set_pixel(x, z, Color(1.0, 1.0 - o, 1.0 - o) if o > 0.0 else (Color(0.3, 0.3, 1.0) if sv > 0.0 else Color(1, 1, 1)))
		out.generate_mipmaps()
	var shtex := ImageTexture.create_from_image(out)
	if gnode != null and gnode.material_override is ShaderMaterial and EV.get("sgnd", "1") == "1": # s231 fix
		var gm := gnode.material_override as ShaderMaterial
		gm.set_shader_parameter("use_shade", true)
		gm.set_shader_parameter("shade_tex", shtex)
		gm.set_shader_parameter("shade_ext", ext)
	var mi := MeshInstance3D.new()
	mi.name = "S230Shade"
	var pm := PlaneMesh.new()
	pm.size = Vector2(2.0 * ext, 2.0 * ext)
	mi.mesh = pm
	var sh := Shader.new()
	sh.code = """shader_type spatial;
render_mode blend_mul, unshaded, depth_draw_never, cull_disabled, shadows_disabled;
uniform sampler2D shade : filter_linear_mipmap;
void fragment() { ALBEDO = texture(shade, UV).rgb; }
"""
	var mat := ShaderMaterial.new()
	mat.shader = sh
	mat.set_shader_parameter("shade", shtex)
	mi.material_override = mat
	st.add_child(mi)
	mi.global_position = Vector3(0.0, float(EV.get("sy", "0.02")), 0.0)
	print("s230 plane xf ", mi.global_transform, " st ", (st as Node3D).global_transform)
	print("s230 shade: verts ", n_v, " sun elev tan ", elev_tan, " dir ", horiz)


func _s230_blur(im: Image, r: float) -> Image:
	if r <= 0.0:
		return im
	var w := im.get_width()
	var small := maxi(8, int(w / maxf(r, 1.0)))
	var b := im.duplicate() as Image
	b.resize(small, small, Image.INTERPOLATE_BILINEAR)
	b.resize(w, w, Image.INTERPOLATE_CUBIC)
	return b


# s231 (FF Re: grass cards along the ground edge and at the foot of every rock). Args: rgn rim count · rgw0/rgw1 rim
# width · sgn tufts per stone · sgw0/sgw1 stone tuft width · rgt tint · rgin/rgout rim band (x disc radius).
func _s231_grass(st: Node, EV: Dictionary) -> void:
	var g := st.find_child("S230Ground", true, false) as MeshInstance3D
	if g == null:
		print("s231 grass: needs gnd=1")
		return
	var r := (g.mesh as CylinderMesh).top_radius
	var top := g.global_position.y + 0.15
	var c := Vector2(g.global_position.x, g.global_position.z)
	var tt: Array = str(EV.get("rgt", "0.55,0.66,0.5")).split_floats(",")
	var tint := Color(tt[0], tt[1], tt[2])
	var files := ["grass0", "grass1", "grass2", "grass3", "grass4", "grass5"]
	var rng := RandomNumberGenerator.new()
	rng.seed = 23101
	var gy := func(p: Vector2) -> float: return top if p.distance_to(c) < r else 0.0
	var n_rim := int(EV.get("rgn", "140"))
	for i in n_rim:
		var a := rng.randf_range(0.0, TAU)
		var rr := r * rng.randf_range(float(EV.get("rgin", "0.97")), float(EV.get("rgout", "1.05")))
		var p := c + Vector2(cos(a), sin(a)) * rr
		_card(st, "res://art/s217/%s.png" % files[rng.randi() % files.size()], Vector3(p.x, gy.call(p), p.y), rng.randf_range(float(EV.get("rgw0", "0.4")), float(EV.get("rgw1", "0.9"))), tint * rng.randf_range(0.85, 1.05))
	var n_st := 0
	for n in st.find_children("*", "", true, false):
		var nm := String(n.name)
		if not (nm.begins_with("Stone") or nm.begins_with("Fallen") or nm.begins_with("Rock")):
			continue
		if n.get_parent() != null and (String(n.get_parent().name).begins_with("Stone") or String(n.get_parent().name).begins_with("Fallen") or String(n.get_parent().name).begins_with("Rock")):
			continue
		var box := AABB()
		var first := true
		for mi in n.find_children("*", "MeshInstance3D", true, false):
			var m := mi as MeshInstance3D
			if m.mesh == null or not m.is_visible_in_tree():
				continue
			var b: AABB = m.global_transform * m.get_aabb()
			box = b if first else box.merge(b)
			first = false
		if first:
			continue
		n_st += 1
		var cc := Vector2(box.get_center().x, box.get_center().z)
		var hx := box.size.x * 0.5
		var hz := box.size.z * 0.5
		for k in int(EV.get("sgn", "7")):
			var a := rng.randf_range(0.0, TAU)
			var p := cc + Vector2(cos(a) * hx, sin(a) * hz) * rng.randf_range(0.8, 1.05)
			_card(st, "res://art/s217/%s.png" % files[rng.randi() % files.size()], Vector3(p.x, gy.call(p), p.y), rng.randf_range(float(EV.get("sgw0", "0.35")), float(EV.get("sgw1", "0.8"))), tint * rng.randf_range(0.8, 1.0))
	print("s231 grass: rim ", n_rim, " stones ", n_st)
