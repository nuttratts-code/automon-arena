// s231 FF Re-style grass patch (own code, numbers from hd-2d-ref): small leaning blade cards, normals = straight up
// (lit exactly like the ground), vertex colour R 0 root -> 1 tip (root darkening + wind mask), baked shade by world xz.
shader_type spatial;
render_mode cull_disabled, depth_draw_opaque;
uniform sampler2D tex : source_color, filter_linear_mipmap_anisotropic;
uniform vec3 tint = vec3(1.0);
uniform float bright = 1.0;
uniform float root_dark = 0.55;
uniform float wind = 0.04;
uniform float wind_speed = 0.9;
uniform bool use_shade = false;
uniform sampler2D shade_tex : filter_linear_mipmap;
uniform float shade_ext = 26.0;
varying vec3 wp;
void vertex() {
	wp = (MODEL_MATRIX * vec4(VERTEX, 1.0)).xyz;
	float w = sin(TIME * wind_speed + wp.x * 0.35 + wp.z * 0.21) * 0.6 + sin(TIME * wind_speed * 2.3 + wp.x * 0.9) * 0.4;
	VERTEX.xz += vec2(1.0, 0.3) * w * wind * COLOR.r * COLOR.r;
}
void fragment() {
	vec4 c = texture(tex, UV);
	ALPHA = c.a;
	ALPHA_SCISSOR_THRESHOLD = 0.5;
	vec3 col = c.rgb * tint * bright * mix(root_dark, 1.0, COLOR.r);
	if (use_shade) { col *= texture(shade_tex, (wp.xz + shade_ext) / (2.0 * shade_ext)).rgb; }
	ALBEDO = col;
	NORMAL = normalize((VIEW_MATRIX * vec4(0.0, 1.0, 0.0, 0.0)).xyz); // both faces lit like the ground (cull_disabled flips back faces)
	ROUGHNESS = 1.0;
	SPECULAR = 0.0;
}
