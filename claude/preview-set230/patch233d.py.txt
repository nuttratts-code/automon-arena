import sys, os
T = sys.argv[1]
def patch(p, pairs, tail=''):
    p = os.path.join(T, p)
    s = open(p, encoding='utf-8').read()
    for a, b in pairs:
        assert s.count(a) == 1, (p, a[:70])
        s = s.replace(a, b)
    s += tail
    open(p, 'w', encoding='utf-8').write(s)

# s233 R11 (owner G2): lower camera like the owner painting. Sprites re-lean to the new pitch, feet stay put.
func _s233_cam(st: Node, cam: Camera3D, sprs: Array, EV: Dictionary) -> void:
	var p := deg_to_rad(float(EV["cpitch"]))
	var cp: Array = str(EV.get("cpos", "0,6.0,29.4")).split_floats(",")
	cam233 = Transform3D(Basis.from_euler(Vector3(-p, 0, 0)), Vector3(cp[0], cp[1], cp[2]))
	cam.transform = cam233
	var ol: Vector3 = st.lean()
	var nl := Vector3(0.0, cos(p), -sin(p))
	for s in sprs:
		var sp := s as Sprite3D
		var hh := sp.pixel_size * float(sp.texture.get_height()) * 0.5
		var feet := sp.position - ol * hh
		sp.position = feet + nl * hh
		sp.rotation.x = -p
	print("s233 cam pitch ", EV["cpitch"], " pos ", cam233.origin)

# Their stage shadow cards ("Shadow" + name) move by the same amount.
const RING233 := {
	"StoneL": Vector2(-12.5, -9.5), "StoneBig": Vector2(-9.2, -12.4), "StoneBigSide": Vector2(-7.4, -13.8),
	"StoneMid": Vector2(-4.5, -14.0), "StoneR": Vector2(10.0, -11.5), "StoneFrontR": Vector2(12.8, 5.0),
	"StoneFrontL": Vector2(-12.9, 4.5), "StoneShortR": Vector2(-14.8, -2.5), "FallenSlab": Vector2(6.0, -15.0),
	"FallenSlabR": Vector2(10.6, 9.6), "RockSlab": Vector2(3.4, -15.4), "RockFarR": Vector2(8.6, -14.6),
	"RockFrontL": Vector2(-10.4, 8.2), "RockFarL": Vector2(-17.5, -8.0),
}
func _s233_ring(st: Node, EV: Dictionary) -> void:
	var n_mv := 0
	for nm in RING233.keys():
		var n := st.find_child(nm, true, false) as Node3D
		if n == null:
			continue
		var to: Vector2 = RING233[nm]
		var d := Vector3(to.x - n.global_position.x, 0, to.y - n.global_position.z)
		n.global_position += d
		var sh := st.find_child("Shadow" + nm, true, false) as Node3D
		if sh != null:
			sh.global_position += d
		n_mv += 1
	print("s233 ring: moved ", n_mv)

# s233 R11: the near hills card goes (3D ridges replace it); the arch stands on a ridge top right of centre.
func _s233_far(st: Node, EV: Dictionary) -> void:
	for mi in st.find_children("*", "MeshInstance3D", true, false):
		var m := (mi as MeshInstance3D).material_override as StandardMaterial3D
		if m != null and m.albedo_texture != null and m.albedo_texture.resource_path.ends_with("far_hills_v3.png"):
			var z := (mi as Node3D).global_position.z
			if z > -150.0:
				(mi as Node3D).visible = EV.get("hc1", "0") == "1"
			else:
				(mi as Node3D).global_position.y += float(EV.get("hc2y", "0"))
				(mi as Node3D).visible = EV.get("hc2", "1") == "1"
	var arch := st.find_child("LandmarkArch", true, false) as MeshInstance3D
	if arch != null:
		var ap: Array = str(EV.get("arch", "34,-125,9")).split_floats(",")
		var bb := arch.mesh.get_aabb()
		var ks := ap[2] / maxf(bb.size.y, 0.01)
		arch.scale = Vector3.ONE * ks
		var y := _s233_th(Vector2(ap[0], ap[1]), EV)
		arch.global_position = Vector3(ap[0], y - 0.6 - bb.position.y * ks, ap[1]) - Vector3(bb.get_center().x, 0, bb.get_center().z) * ks
		print("s233 arch at ", arch.global_position, " ground ", y)
