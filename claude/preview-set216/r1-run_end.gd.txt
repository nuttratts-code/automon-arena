static func _n(x: Variant) -> String:
	var f := float(x)
	return str(int(f)) if f == floorf(f) else str(f)

static func _k(x: Variant) -> String:
	var s := str(int(float(x)))
	var o := ""
	while s.length() > 3:
		o = "," + s.right(3) + o
		s = s.left(s.length() - 3)
	return s + o

func _h3(t: String) -> Label:
	return TH.label(t, 15, "ink", "display_b")

func _wrap(t: String, col: String = "ink", sz: int = 13) -> Label:
	var l := TH.label(t, sz, col)
	l.autowrap_mode = TextServer.AUTOWRAP_WORD_SMART
	return l

func _sheet_box(alpha: float) -> StyleBoxFlat:
	var s := TH.box(Color(TH.c("panel"), alpha), TH.c("line"), 7, 1, 16, 12)
	TH.ledge(s, false)
	return s

func show_details(on: bool) -> void:
	details.visible = on
	btn_more.text = "ปิดสถิติ" if on else "สถิติ · รีเพลย์"

func _clear(b: Node) -> void:
	for c in b.get_children():
		b.remove_child(c)
		c.queue_free()

func _name_of(id: Variant) -> String:
	var c: Variant = app.cm.get(id)
	return "%s %s" % [id, c["name"]] if c is Dictionary else str(id)

func _tile(num: String, lab: String, col: String) -> PanelContainer:
	var p := PanelContainer.new()
	p.size_flags_horizontal = Control.SIZE_EXPAND_FILL
	p.add_theme_stylebox_override("panel", TH.box(Color(TH.c("bg"), 0.55), Color(0, 0, 0, 0), 6, 0, 8, 4))
	var v := VBoxContainer.new()
	v.add_theme_constant_override("separation", -4)
	var n := TH.label(num, 26, col, "display_b")
	n.horizontal_alignment = HORIZONTAL_ALIGNMENT_CENTER
	v.add_child(n)
	var l := TH.label(lab, 13, "mute")
	l.horizontal_alignment = HORIZONTAL_ALIGNMENT_CENTER
	v.add_child(l)
	p.add_child(v)
	return p

func render() -> void:
	var P: Dictionary = app.P
	var S: Dictionary = R.RULES()["shop"]
	var last := float(R.js_i32(RB.RUN_STAGE()["settings"].get("lastStage")))
	var won := float(P["wins"]) >= float(S["winsToFinish"])
	var is_extra := SH._truthy(P.get("extra"))
	var can_extra := won and not is_extra and float(P["hearts"]) > 0.0 and float(P["round"]) < last
	var all_rounds := float(P["hearts"]) > 0.0 and float(P["round"]) >= last
	title.text = ("ชนะครบ %s แมตช์" % _n(S["winsToFinish"])) if won and not is_extra and not all_rounds else TH.ui_text("end_all", {"n": _n(P["round"])}) if all_rounds \
		else TH.ui_text("end_extra_done", {"n": _n(P["wins"])}) if is_extra else "จบ run · แพ้"
	title.add_theme_color_override("font_color", TH.c("ok" if won or all_rounds else "bad"))
	summary.text = "หัวใจเหลือ %s/%s · เล่น %s รอบ" % [_n(P["hearts"]), _n(S["hearts"]), _n(P["round"])] \
		+ (TH.ui_text("r_dummy", {"n": _n(app.dummy_dmg)}) if app.dummy_dmg > 0.0 else "")
	player.text = TH.ui_text("player", {"n": app.player_name if app.player_name != "" else TH.ui_text("no_name")})
	extra_hint.visible = can_extra
	extra_hint.text = "โหมดพิเศษ: ศัตรูแรงขึ้น %s%% ต่อรอบ · เล่นได้ถึงรอบ %s · หัวใจหมด = จบ" % [_n(float(R.js_i32(RB.RUN_STAGE()["settings"].get("foeExtraPct")))), _n(last)]
	btn_extra.visible = can_extra
	for k in ["normal", "hover", "pressed", "disabled"]:
		btn_new.remove_theme_stylebox_override(k)
	for k in ["font_color", "font_hover_color", "font_pressed_color", "font_focus_color"]:
		btn_new.remove_theme_color_override(k)
	btn_new.remove_theme_font_override("font")
	btn_new.remove_theme_font_size_override("font_size")
	btn_new.custom_minimum_size.y = 44
	if not can_extra:
		TH.primary(btn_new, true)
	var st: Dictionary = app.stats
	_clear(stat_row)
	stat_row.add_child(_tile(_n(st["w"]), "ชนะ", "ok"))
	stat_row.add_child(_tile(_n(st["l"]), "แพ้", "bad"))
	stat_row.add_child(_tile(_n(st["best"]), "ชนะติดกัน", "ink"))
	var ranked: Array = (st["dmg"] as Dictionary).keys()
	ranked.sort_custom(func(a, b): return float(st["dmg"][a]) > float(st["dmg"][b]))
	_clear(mvp_row)
	mvp_row.visible = not ranked.is_empty()
	if not ranked.is_empty():
		var c: Variant = app.cm.get(ranked[0])
		if c is Dictionary:
			var mon := UV.make(c, 1, 56)
			mon.size_flags_vertical = Control.SIZE_SHRINK_CENTER
			mvp_row.add_child(mon)
		var v := VBoxContainer.new()
		v.add_theme_constant_override("separation", -2)
		v.add_child(TH.label("ตัวเด่นของ run", 13, "mute"))
		v.add_child(TH.label("%s · ดาเมจ %s" % [str(c["name"]) if c is Dictionary else str(ranked[0]), _k(st["dmg"][ranked[0]])], 15, "ink", "body_b"))
		mvp_row.add_child(v)
	code_msg.text = TH.ui_text("end_code_hint")
	# details (unchanged content, below the fold)
	_clear(stats_box)
	var avg := float(st["timeMs"]) / float(st["matches"]) / 1000.0 if float(st["matches"]) > 0.0 else 0.0
	var lines := [TH.ui_text("end_st1", {"m": _n(st["matches"]), "w": _n(st["w"]), "d": _n(st["d"]), "l": _n(st["l"])}),
		TH.ui_text("end_st2", {"b": _n(st["best"]), "a": "%.1f" % avg})]
	for i in range(0, mini(3, ranked.size())):
		lines.append(TH.ui_text("end_rank", {"i": str(i + 1), "n": _name_of(ranked[i]), "v": _n(st["dmg"][ranked[i]])}))
	for t in lines:
		stats_box.add_child(_wrap("• " + t))
	(details.find_child("EndRepTitle", true, false) as Label).text = TH.ui_text("rep_title", {"n": str(app.replays.size())})
	_clear(rep_box)
	if app.replays.is_empty():
		rep_box.add_child(TH.label(TH.ui_text("rep_none"), 13, "mute"))
	for i in app.replays.size():
		var x: Dictionary = app.replays[i]
		var h := HBoxContainer.new()
		var res := TH.ui_text("res_w" if R.js_seq(x["w"], 0.0) else "res_l" if R.js_seq(x["w"], 1.0) else "res_d")
		var l := TH.label(TH.ui_text("rep_row", {"r": _n(x["r"]), "res": res, "t": "%.1f" % (float(x["t"]) / 1000.0)}), 13, "ink")
		l.size_flags_horizontal = Control.SIZE_EXPAND_FILL
		h.add_child(l)
		var b := Button.new()
		b.name = "EndWatch%d" % i
		b.text = TH.ui_text("rep_watch")
		b.custom_minimum_size.y = 32
		b.pressed.connect(func() -> void: watch.emit(i))
		h.add_child(b)
		rep_box.add_child(h)
	_clear(hist_box)
	var hist: Array = SV.read_history(SV.load_text(app.history_file))
	if hist.is_empty():
		hist_box.add_child(TH.label(TH.ui_text("end_hist_none"), 13, "mute"))
	for hh in hist:
		var d := Time.get_datetime_dict_from_unix_time(int(float(hh["at"]) / 1000.0))
		var tm := []
		for u in hh["team"]:
			tm.append("%s★%s" % [u["id"], _n(u["level"])])
		hist_box.add_child(_wrap(TH.ui_text("end_hrow", {"d": "%d/%d/%d" % [d["day"], d["month"], d["year"] + 543], "res": TH.ui_text("end_h_won" if bool(hh["won"]) else "end_h_lost"),
			"w": _n(hh["wins"]), "r": _n(hh["rounds"]), "t": " ".join(PackedStringArray(tm)) if not tm.is_empty() else "-"})))

func say_code(t: String, err: bool = false) -> void:
	code_msg.visible = true
	code_msg.text = t
	code_msg.add_theme_color_override("font_color", TH.c("bad" if err else "ok"))
