var app: Node

func _frames() -> void:
	for k in 4:
		await process_frame
	await RenderingServer.frame_post_draw

func _initialize() -> void:
	var args := OS.get_cmdline_user_args()
	var out: String = args[args.find("--out") + 1]
	var look: String = args[args.find("--look") + 1] if args.has("--look") else "A"
	var sat: float = float(args[args.find("--sat") + 1]) if args.has("--sat") else 1.15
	var csat: float = float(args[args.find("--csat") + 1]) if args.has("--csat") else 1.0 # characters only
	var EV := {}
	for a in args:
		if a.begins_with("--e:"):
			var kv := a.substr(4).split("=")
			EV[kv[0]] = kv[1]
	var nf: int = int(args[args.find("--frames") + 1]) if args.has("--frames") else 1
	DirAccess.make_dir_recursive_absolute(out)
	await process_frame
	app = load("res://scenes/main.tscn").instantiate()
	app.set_meta("no_boot", true)
	app.set_meta("force_light", true)
	root.add_child(app)
	app.zoom_s = 0.0
	await _frames()
	app.name_scr.visible = false
	app.player_name = "ทดสอบ"
	app.start_new_run(4242)
	app.do_buy(0, null)
	app.do_move({"at": "bench", "i": 0.0}, {"at": "board", "row": 0.0, "lane": 0.0})
	app.set_speed(1, false)
	app._on_fight()
	await _frames()
	app.battle.view.advance(1500.0)
	await _frames()
	var st: Node = app.stage
	for id in (st.groups.get("battle", {}) as Dictionary).keys():
		st.remove_unit("battle", id)
	# you (left): front row 0 → back row 2; foe mirrors the same six
	var ids := ["HL-1", "WN-1", "PL-1", "SF-1", "WY-1", "PT-1"]
	var foe := ["SF-1", "WY-1", "HL-1", "PT-1", "WN-1", "PL-1"]
	var n := 9000
	var map := {}
	for s in 2:
		for row in 3:
			for lane in 2:
				var cid: String = (ids if s == 0 else foe)[row * 2 + lane]
				n += 1
				map[n] = cid
				st.add_unit("battle", {"id": n, "side": s, "row": row, "lane": lane, "c": {"id": "HL-1", "tribe": ""}})
	st.show_group("battle")
	await _frames()
	var cam := root.get_viewport().get_camera_3d()
	# one pixel size for every unit = pixel-perfect at the board centre depth (hd-2d-ref: one UUPerPixel for all)
	var d0: float = (Vector3.ZERO - cam.global_position).dot(-cam.global_transform.basis.z)
	var wpp: float = 2.0 * d0 * tan(deg_to_rad(cam.fov) / 2.0) / float(DisplayServer.window_get_size().y)
	var sprs := []
	var k := 0
	for id in map.keys():
		var s: Sprite3D = st.groups["battle"][id]["spr"]
		var old_tall: float = s.pixel_size * float(s.texture.get_height())
		var img := Image.load_from_file(ProjectSettings.globalize_path("res://tests/s225/%s.png" % map[id]))
		if csat < 1.0: # desaturate the character only, the set keeps its colours
			for y in img.get_height():
				for x in img.get_width():
					var c := img.get_pixel(x, y)
					if c.a > 0.0:
						var l := c.r * 0.3 + c.g * 0.59 + c.b * 0.11
						img.set_pixel(x, y, Color(lerpf(l, c.r, csat), lerpf(l, c.g, csat), lerpf(l, c.b, csat), c.a))
		s.texture = ImageTexture.create_from_image(img)
		s.hframes = 8
		s.pixel_size = wpp
		var new_tall: float = wpp * float(img.get_height())
		s.position += st.lean() * ((new_tall - old_tall) / 2.0)
		s.frame = (k * 3) % 8
		if look != "A":
			s.modulate = Color(0.97, 0.96, 0.93) # scene tint: sprites sit in the same light as the set
			if look == "C":
				s.shaded = true # takes the sun + sky ambient like the set (hd-2d-ref 2.3 minimum)
				s.modulate = Color.WHITE
			# contact shadow: darker core + soft rim (hd-2d-ref: sprites grounded by a foot shadow)
			var sh: MeshInstance3D = st.groups["battle"][id]["shadow"]
			var g := Gradient.new()
			g.offsets = PackedFloat32Array([0.0, 0.45, 1.0])
			g.colors = PackedColorArray([Color(0.05, 0.06, 0.1, 0.7), Color(0.05, 0.06, 0.1, 0.42), Color(0.05, 0.06, 0.1, 0.0)])
			var t := GradientTexture2D.new()
			t.gradient = g; t.fill = GradientTexture2D.FILL_RADIAL
			t.fill_from = Vector2(0.5, 0.5); t.fill_to = Vector2(1.0, 0.5); t.width = 64; t.height = 64
			(sh.material_override as StandardMaterial3D).albedo_texture = t
			sh.scale = Vector3(1.15, 1, 1.0)
		sprs.append(s)
		k += 1
	if look != "A":
		var env: Environment = st.env_node.environment
		env.fog_depth_begin = 10.0      # hd-2d-ref: thin air starts 10 m from the camera
		env.fog_depth_curve = 2.2
		env.fog_light_color = Color("#b7c8da")
		env.tonemap_mode = Environment.TONE_MAPPER_FILMIC
		st.sun.light_energy = 1.25
		st.sun.light_color = Color("#fff4dc")
		var hd: ColorRect = app.get_node("HD2D")
		var m: ShaderMaterial = hd.material
		m.set_shader_parameter("warm", Vector3(1.02, 1.0, 0.96))
		m.set_shader_parameter("sat", sat)
		m.set_shader_parameter("contrast", 1.05)
		m.set_shader_parameter("lift", Vector3(0.006, 0.012, 0.02))
		m.set_shader_parameter("bloom_all", 0.6)
		m.set_shader_parameter("glow", 0.35) # cyan rune bloom was blowing the pale ghost out
		m.set_shader_parameter("vignette", 0.32)
		m.set_shader_parameter("focus_top", 0.22)
		m.set_shader_parameter("focus_bottom", 0.88)
		m.set_shader_parameter("soft", 0.22)
		m.set_shader_parameter("max_lod", 1.6)
	if look == "D" or look == "E" or look == "F": # B4: the real screenshot colours (hd-2d-ref 7.5): pale, soft, cool shadows, teal-grey haze, grey-blue sky
		var env2: Environment = st.env_node.environment
		env2.fog_light_color = Color("#b4cfd6") # teal-grey haze (ref far hills 103,156,152)
		env2.fog_density = 0.4
		env2.fog_depth_curve = 2.0
		for mi in st.find_children("*", "MeshInstance3D", true, false):
			var mm := (mi as MeshInstance3D).material_override as StandardMaterial3D
			if mm != null and mm.albedo_texture != null and mm.albedo_texture.resource_path.ends_with("sky_clouds.png"):
				mm.albedo_color = Color(0.74, 0.83, 0.88) # pale grey-blue like the ref sky (170, 195, 202)
		var m2: ShaderMaterial = (app.get_node("HD2D") as ColorRect).material
		m2.set_shader_parameter("sat", sat if args.has("--sat") else 0.85)
		m2.set_shader_parameter("contrast", 0.97)
		m2.set_shader_parameter("warm", Vector3(1.03, 1.01, 0.97))
		m2.set_shader_parameter("bloom_all", 0.35)
		m2.set_shader_parameter("glow", 0.3)
		m2.set_shader_parameter("vignette", 0.3)
	if look == "E" or look == "F": # B5: background only, matched to the ref screenshot's measured colours (hd-2d-ref 7.5 + compare shot)
		var e: Environment = st.env_node.environment
		e.fog_light_color = Color(EV.get("fog", "#8fb3ae"))
		e.fog_density = float(EV.get("fogd", "0.5"))
		e.ambient_light_sky_contribution = float(EV.get("amb", "0.8"))
		st.sun.light_energy = float(EV.get("sun", "0.95"))
		var gt: Array = str(EV.get("ground", "0.56,0.7,0.6")).split_floats(",")
		var sk: Array = str(EV.get("sky", "0.6,0.7,0.76")).split_floats(",")
		for mi in st.find_children("*", "MeshInstance3D", true, false):
			var mo: Material = (mi as MeshInstance3D).material_override
			if mo is ShaderMaterial and (mo as ShaderMaterial).get_shader_parameter("tint") != null:
				(mo as ShaderMaterial).set_shader_parameter("tint", Vector3(gt[0], gt[1], gt[2])); print("TINT ", mi.name)
			elif mo is StandardMaterial3D and (mo as StandardMaterial3D).albedo_texture != null and (mo as StandardMaterial3D).albedo_texture.resource_path.ends_with("sky_clouds.png"):
				(mo as StandardMaterial3D).albedo_color = Color(sk[0], sk[1], sk[2])
			elif mo is StandardMaterial3D and (mo as StandardMaterial3D).albedo_texture != null and (mo as StandardMaterial3D).albedo_texture.resource_path.ends_with("far_hills_v3.png"):
				var hl: Array = str(EV.get("hills", "0.55,0.75,0.72")).split_floats(",")
				var far: bool = (mi as Node3D).position.z < -150.0
				(mo as StandardMaterial3D).albedo_color = Color(hl[0], hl[1], hl[2]) * (0.9 if far else 1.0)
		var m3: ShaderMaterial = (app.get_node("HD2D") as ColorRect).material
		m3.set_shader_parameter("sat", float(EV.get("psat", "0.9")))
		m3.set_shader_parameter("contrast", float(EV.get("pcon", "1.02")))
		m3.set_shader_parameter("vignette", float(EV.get("vig", "0.42")))
	if look == "F": # B6: scene layers like the ref, owner's own pieces only (grass cards art/s217, stones, cliff, slab)
		_b6(st, cam, EV)
	await _frames()
	print("S225 look ", look, " wpp ", wpp, " cam ", cam.global_position)
	for f in nf:
		for s in sprs:
			(s as Sprite3D).frame = ((s as Sprite3D).frame + (1 if f > 0 else 0)) % 8
		await _frames()
		root.get_texture().get_image().save_png(out.path_join("%s_s%s_c%s_f%02d.png" % [look, str(sat), str(csat), f]))
	quit()


func _card(st: Node, file: String, at: Vector3, width: float, tint: Color) -> void:
	var t: Variant = load(file)
	if not (t is Texture2D):
		return
	var sp := Sprite3D.new()
	sp.texture = t
	sp.pixel_size = width / float((t as Texture2D).get_width())
	sp.billboard = BaseMaterial3D.BILLBOARD_FIXED_Y
	sp.alpha_cut = SpriteBase3D.ALPHA_CUT_DISCARD
	sp.shaded = false
	sp.modulate = tint
	sp.cast_shadow = GeometryInstance3D.SHADOW_CASTING_SETTING_OFF
	sp.position = at + Vector3(0, width * float((t as Texture2D).get_height()) / float((t as Texture2D).get_width()) * 0.5 - 0.12, 0)
	st.add_child(sp)

func _b6(st: Node, cam: Camera3D, EV: Dictionary) -> void:
	# 1 ground: shift the yellow-green toward the ref's blue-green (74,127,111) — a tint cannot add blue the texture lacks
	var teal := float(EV.get("teal", "0.5"))
	for mi in st.find_children("*", "MeshInstance3D", true, false):
		var mo: Material = (mi as MeshInstance3D).material_override
		if mo is ShaderMaterial and (mo as ShaderMaterial).get_shader_parameter("tint") != null:
			var sh := Shader.new()
			sh.code = (mo as ShaderMaterial).shader.code.replace("ALBEDO = c * tint;", "float l = dot(c, vec3(0.3, 0.59, 0.11));
	c = mix(c, l * vec3(0.78, 1.32, 1.18), %s);
	ALBEDO = c * tint;" % str(teal))
			(mo as ShaderMaterial).shader = sh
	# 2 plaza: warm beige like the ref's dirt path, a bit darker (owner tone script on the Plaza node)
	for n in st.find_children("Plaza", "", true, false):
		if "tone" in n:
			var pt: Array = str(EV.get("plaza", "0.66,0.6,0.48")).split_floats(",")
			n.tone = Color(pt[0], pt[1], pt[2])
	# 3 grass tufts: a ring at the plaza rim (back + sides), never inside the board
	var rng := RandomNumberGenerator.new()
	rng.seed = 22501
	var files := ["grass0", "grass1", "grass2", "grass3", "grass4", "grass5", "grass_big"]
	var gtint := Color(0.62, 0.74, 0.66)
	var ring_n := int(EV.get("tufts", "70"))
	for i in ring_n:
		var a := rng.randf_range(PI * 0.92, PI * 2.08) # back half + the two sides
		var r := rng.randf_range(15.0, 21.0)
		var at := Vector3(cos(a) * r * 1.12, 0, sin(a) * r - 1.5)
		var w := rng.randf_range(float(EV.get("tw0", "0.7")), float(EV.get("tw1", "1.4")))
		_card(st, "res://art/s217/%s.png" % files[rng.randi() % files.size()], at, w, gtint * rng.randf_range(0.85, 1.05))
	# 4 foreground framing: big tufts at the bottom corners, soft in the tilt-shift band (ref: grass between camera and field)
	if EV.get("fg", "1") == "1": # the view's bottom edge meets the ground at z ~7, half-width ~6 m there
		for fx in [[-7.4, 7.6, 2.2], [-5.6, 8.4, 1.6], [7.4, 7.8, 2.1], [5.8, 8.6, 1.5]]:
			_card(st, "res://art/s217/%s.png" % files[rng.randi() % files.size()], Vector3(fx[0], 0, fx[1]), fx[2], gtint * 0.72)
